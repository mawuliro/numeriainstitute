    25→def extraire_id_youtube(url: str | None):
    26→    """Extract YouTube video ID from common URL formats."""
    27→    if not url or 'vimeo' in (url or '').lower():
    28→        return None
    29→    url = url.strip()
    30→    if 'youtu.be/' in url:
    31→        return url.split('youtu.be/')[-1].split('?')[0].strip('/')
    32→    if '/shorts/' in url:
    33→        return url.split('/shorts/')[-1].split('?')[0].strip('/')
    34→    if '/embed/' in url:
    35→        return url.split('/embed/')[-1].split('?')[0].strip('/')
    36→    if 'v=' in url:
    37→        return url.split('v=')[-1].split('&')[0].strip()
    38→    return None
    39→
    40→
    41→def convertir_url_youtube(url: str | None):
    42→    """Convert YouTube/Vimeo URL to embed URL (display-time only)."""
    43→    if not url:
    44→        return url
    45→
    46→    url = url.strip()
    47→    u = url.lower()
    48→
    49→    # Vimeo
    50→    if 'vimeo.com' in u:
    51→        if '/video/' in u:
    52→            video_id = url.split('/video/')[-1].split('?')[0].split('/')[0]
    53→        else:
    54→            video_id = url.split('vimeo.com/')[-1].split('?')[0].split('/')[0]
    55→        if video_id.isdigit():
    56→            return f'https://player.vimeo.com/video/{video_id}?color=E8A020&title=0&byline=0'
    57→        return url
    58→
    59→    # YouTube
    60→    if 'youtube-nocookie.com/embed/' in u:
    61→        return url
    62→    if 'youtube.com/embed/' in u:
    63→        video_id = url.split('/embed/')[-1].split('?')[0].strip('/')
    64→        return f'https://www.youtube-nocookie.com/embed/{video_id}?rel=0&modestbranding=1'
    65→    if 'youtu.be/' in u:
    66→        video_id = url.split('youtu.be/')[-1].split('?')[0].strip('/')
    67→        return f'https://www.youtube-nocookie.com/embed/{video_id}?rel=0&modestbranding=1'
    68→    if '/shorts/' in u:
    69→        video_id = url.split('/shorts/')[-1].split('?')[0].strip('/')
    70→        return f'https://www.youtube-nocookie.com/embed/{video_id}?rel=0&modestbranding=1'
    71→    if 'youtube.com' in u and 'v=' in url:
    72→        video_id = url.split('v=')[-1].split('&')[0].strip()
    73→        return f'https://www.youtube-nocookie.com/embed/{video_id}?rel=0&modestbranding=1'
    74→    return url

   129→    class Meta:

   132→
   133→    def __str__(self):
   134→        return self.title
   135→
   136→    def save(self, *args, **kwargs):
   137→        if not self.slug:
   138→            base = slugify(self.title)
   139→            slug = base
   140→            n = 1
   141→            while Course.objects.filter(slug=slug).exclude(pk=self.pk).exists():
   142→                slug = f'{base}-{n}'
   143→                n += 1
   144→            self.slug = slug
   145→        super().save(*args, **kwargs)

   258→    def get_nom_fichier(self):
   259→        """Return a safe filename for the PDF certificate download."""
   260→        nom = slugify(self.inscription.etudiant.get_full_name() or self.inscription.etudiant.username)
   261→        cours = slugify(self.inscription.course.title)
   262→        return f'certificat-{nom}-{cours}.pdf'

   265→        ordering = ['-date_emission']

   381→    def get_payload(self, user=None, preview=False):
   382→        """Build a sanitised dict for the lesson_blocks_render.html template.
   383→
   384→        Never includes solution_code or test_code in plain text —
   385→        test_code is base64-encoded for client-side evaluation only.
   386→        """
   387→        import base64
   388→
   389→        data = {
   390→            'id': self.id,
   391→            'type': self.block_type,
   392→            'order': self.order,
   393→        }
   394→
   395→        if self.block_type == 'text':
   396→            data['text_content'] = self.text_content or ''
   397→
   398→        elif self.block_type == 'video':
   399→            data['embed_url'] = self._embed_url(self.video_url)
   400→            data['video_caption'] = self.video_caption or ''
   401→
   402→        elif self.block_type == 'sandbox':
   403→            data['title'] = self.sandbox_title or 'Essaie toi-même'
   404→            data['initial_code'] = self.sandbox_initial_code or ''
   405→
   406→        elif self.block_type == 'code_exercise':
   407→            ex = self.code_exercise
   408→            if ex:
   409→                data['exercise_id'] = ex.id
   410→                data['id'] = ex.id
   411→                data['title'] = ex.title
   412→                data['instructions'] = ex.instructions or ''
   413→                data['difficulty'] = ex.difficulty
   414→                data['points'] = ex.points
   415→                data['hint'] = ex.hint or ''
   416→                data['max_attempts'] = ex.max_attempts
   417→                data['starter_code'] = ex.starter_code or ''
   418→                data['expected_output'] = ex.expected_output or ''
   419→                data['evaluation_mode'] = ex.evaluation_mode
   420→                data['test_code_b64'] = base64.b64encode(
   421→                    (ex.test_code or '').encode()
   422→                ).decode() if ex.test_code else ''
   423→                data['order'] = ex.order
   424→                data['is_solved'] = False
   425→                data['points_earned'] = 0
   426→                data['attempts_used'] = 0
   427→                if user and not preview:
   428→                    from cours.models import StudentProgress, ExerciseAttempt
   429→                    progress = StudentProgress.objects.filter(
   430→                        student=user, exercise_type='code',
   431→                        exercise_id=ex.id
   432→                    ).first()
   433→                    if progress:
   434→                        data['is_solved'] = progress.is_solved
   435→                        data['points_earned'] = progress.points_earned
   436→                    data['attempts_used'] = ExerciseAttempt.objects.filter(
   437→                        exercise_type='code', exercise_id=ex.id, student=user
   438→                    ).count()
   439→
   440→        elif self.block_type == 'mcq':

   442→            if ex:
   443→                data['mcq_id'] = ex.id
   444→                data['mcq_title'] = ex.title
   445→                data['title'] = ex.title
   446→                data['question'] = ex.question or ''
   447→                data['hint'] = ex.hint or ''
   448→                data['allow_multiple'] = ex.allow_multiple_correct
   449→                data['shuffle'] = ex.shuffle_choices
   450→                data['max_attempts'] = ex.max_attempts
   451→                data['points'] = ex.points
   452→                data['difficulty'] = ex.difficulty
   453→                data['explanation'] = ''  # hidden until solved/exhausted
   454→                data['is_solved'] = False
   455→                data['points_earned'] = 0
   456→                data['attempts_used'] = 0
   457→                data['correct_ids'] = []
   458→                data['choices'] = [
   459→                    {'id': c.id, 'text': c.text, 'order': c.order}
   460→                    for c in ex.choices.all().order_by('order')
   461→                ]
   462→                if user and not preview:
   463→                    from cours.models import StudentProgress, ExerciseAttempt
   464→                    progress = StudentProgress.objects.filter(
   465→                        student=user, exercise_type='mcq',
   466→                        exercise_id=ex.id
   467→                    ).first()

   472→                        exercise_type='mcq', exercise_id=ex.id, student=user

   474→                    # Reveal correct answers if solved or attempts exhausted

   489→                data['hint'] = ex.hint or ''

   496→                        student=user, exercise_type='fill_blank',
   497→                        exercise_id=ex.id
   498→                    ).first()
   499→                    if progress:
   500→                        data['is_solved'] = progress.is_solved
   501→                    data['attempts_used'] = ExerciseAttempt.objects.filter(
   502→                        exercise_type='fill_blank', exercise_id=ex.id, student=user

   504→
   505→        elif self.block_type == 'true_false':
   506→            ex = self.true_false
   507→            if ex:

   509→                data['title'] = ex.title

   511→                data['points_per_statement'] = ex.points_per_statement

   514→                data['hint'] = ex.hint or ''

   527→                        exercise_type='true_false', exercise_id=ex.id, student=user

   529→
   530→        elif self.block_type == 'code_order':
   531→            ex = self.code_order
   532→            if ex:
   533→                data['code_order_id'] = ex.id
   534→                data['title'] = ex.title

   544→        elif self.block_type == 'matching':
   545→            ex = self.matching
   546→            if ex:
   547→                data['matching_id'] = ex.id

   559→            if ex:
   560→                data['short_answer_id'] = ex.id
   561→                data['title'] = ex.title
   562→                data['instructions'] = ex.instructions or ''
   563→                data['points'] = ex.points
   564→                data['difficulty'] = ex.difficulty
   565→                data['hint'] = ex.hint or ''
   566→                data['max_attempts'] = ex.max_attempts
   567→                data['is_solved'] = False
   568→                data['attempts_used'] = 0
   569→
   570→        return data
   571→
   572→    @staticmethod
   573→    def _embed_url(url):
   574→        """Convert YouTube/Vimeo URLs to embeddable iframe URLs."""
   575→        if not url:
   576→            return ''
   577→        u = url.strip()
   578→        if 'youtube.com/watch' in u and 'v=' in u:
   579→            vid = u.split('v=', 1)[1].split('&', 1)[0]
   580→            return f'https://www.youtube.com/embed/{vid}'
   581→        if 'youtu.be/' in u:
   582→            vid = u.split('youtu.be/', 1)[1].split('?', 1)[0]
   583→            return f'https://www.youtube.com/embed/{vid}'
   584→        if 'vimeo.com/' in u and '/embed/' not in u:
   585→            vid = u.rstrip('/').split('/')[-1]
   586→            return f'https://player.vimeo.com/video/{vid}'
   587→        return u
