   100→@staff_only
   101→def contact_detail(request, message_id):
   102→    from pages.models import ContactMessage
   103→    from numeria_project.emails import send_contact_reply_email
   104→
   105→    msg = get_object_or_404(ContactMessage, id=message_id)
   106→
   107→    # Auto-mark as read on open
   108→    if msg.status == 'unread':
   109→        msg.status = 'read'
   110→        msg.is_read = True
   111→        msg.save(update_fields=['status', 'is_read'])
   112→
   113→    if request.method == 'POST':
   114→        to_field = request.POST.get('to', msg.email).strip()
   115→        subject = request.POST.get('subject', f'Re: {msg.subject}').strip()
   116→        reply_text = request.POST.get('message', '').strip()
   117→
   118→        if not reply_text:
   119→            messages.error(request, _("Le message de réponse ne peut pas être vide."))
   120→        else:
   121→            try:
   122→                validate_email(to_field)
   123→                subject = subject.replace('\n', ' ').replace('\r', ' ')
   124→            except ValidationError:
   125→                messages.error(request, _("L'adresse email de destination n'est pas valide."))
   126→                return render(request, 'admin_panel/contact_detail.html', {
   127→                    'msg': msg,
   128→                    'replies': msg.replies.select_related('replied_by').all(),
   129→                    'reply_subject': subject,
   130→                    'reply_to': to_field,
   131→                    'reply_message': reply_text,
   132→                })
   133→            try:
   134→                send_contact_reply_email(
   135→                    to_email=to_field,
   136→                    to_name=msg.name,
   137→                    subject=subject,
   138→                    reply_message=reply_text,
   139→                    staff_name=request.user.get_full_name() or request.user.username,
   140→                )
   141→                ContactReply.objects.create(
   142→                    contact_message=msg,
   143→                    replied_by=request.user,
   144→                    subject=subject,
   145→                    message=reply_text,
   146→                )
   147→                msg.status = 'replied'
   148→                msg.is_read = True
   149→                msg.save(update_fields=['status', 'is_read'])
   150→
   151→                staff_name = request.user.get_full_name() or request.user.username
   152→                notify_group(
   153→                    User.objects.filter(is_staff=True),
   154→                    title=_("Réponse envoyée"),
   155→                    message=f"{staff_name} a répondu à {msg.name}",
   156→                    notification_type='info',
   157→                    link=f'/admin-panel/contacts/{msg.id}/',
   158→                    created_by=request.user,
   159→                )
   160→                log_staff_action(
   161→                    performed_by=request.user,
   162→                    action_type='contact_replied',
   163→                    description=f"Réponse envoyée à {msg.name} ({msg.email}) — sujet: {msg.subject}",
   164→                )
   165→                messages.success(request, _("Réponse envoyée avec succès."))
   166→                return redirect('admin_panel:contacts_list')
   167→            except Exception as e:
   168→                logger.error('contact_detail reply error: %s', e)
   169→                messages.error(request, _("Erreur lors de l'envoi de l'email : %(err)s") % {'err': str(e)})
   170→
   171→    replies = msg.replies.select_related('replied_by').all()
   172→    return render(request, 'admin_panel/contact_detail.html', {
   173→        'msg': msg,
   174→        'replies': replies,
   175→        'reply_subject': f'Re: {msg.subject}',
   176→    })

   231→@staff_only
   232→@require_POST
   233→def candidature_action(request, candidature_id):
   234→    from admissions.models import Candidature
   235→    from numeria_project.emails import (
   236→        send_candidacy_acceptance_email,
   237→        send_candidacy_rejection_email,
   238→    )
   239→
   240→    candidature = get_object_or_404(Candidature, id=candidature_id)
   241→    action = request.POST.get('action')
   242→    notes = request.POST.get('notes', '').strip()
   243→
   244→    if action in ['accept', 'reject'] and candidature.statut in ['acceptee', 'refusee']:
   245→        messages.warning(request, _("Cette candidature a déjà été traitée et ne peut plus être modifiée."))
   246→        return redirect('admin_panel:candidature_detail', candidature_id=candidature_id)
   247→
   248→    if action == 'accept':
   249→        candidature.statut = 'acceptee'
   250→        candidature.commentaire_admin = notes
   251→        candidature.date_decision = timezone.now()
   252→        candidature.reviewed_by = request.user
   253→        candidature.reviewed_at = timezone.now()
   254→        candidature.save()
   255→        try:
   256→            send_candidacy_acceptance_email(candidature)
   257→        except Exception as e:
   258→            logger.error('candidacy accept email failed: %s', e)
   259→        notify_user(
   260→            candidature.utilisateur,
   261→            title=_("Candidature acceptée !"),
   262→            message=_("Votre candidature pour %(prog)s a été acceptée ! 🎉") % {
   263→                'prog': candidature.campagne.filiere
   264→            },
   265→            notification_type='candidacy',
   266→            link='/admissions/mes-candidatures/',
   267→            created_by=request.user,
   268→        )
   269→        log_staff_action(request.user, 'candidacy_accepted',
   270→                         f"Candidature {candidature.reference} acceptée ({candidature.campagne.filiere})",
   271→                         target_user=candidature.utilisateur)
   272→        messages.success(request, _("✅ Candidature de %(name)s acceptée.") % {
   273→            'name': f"{candidature.prenom} {candidature.nom}"})
   274→
   275→    elif action == 'reject':
   276→        rejection_reason = request.POST.get('rejection_reason', '').strip()
   277→        if not rejection_reason:
   278→            messages.error(request, _("La raison de refus est obligatoire."))
   279→            return redirect('admin_panel:candidature_detail', candidature_id=candidature_id)
   280→        candidature.statut = 'refusee'
   281→        candidature.rejection_reason = rejection_reason
   282→        candidature.commentaire_admin = notes
   283→        candidature.date_decision = timezone.now()
   284→        candidature.reviewed_by = request.user
   285→        candidature.reviewed_at = timezone.now()
   286→        candidature.save()
   287→        try:
   288→            send_candidacy_rejection_email(
   289→                candidature.utilisateur,
   290→                candidature.campagne.filiere,
   291→                rejection_reason,
   292→            )
   293→        except Exception as e:
   294→            logger.error('candidacy reject email failed: %s', e)
   295→        notify_user(
   296→            candidature.utilisateur,
   297→            title=_("Résultat de votre candidature"),
   298→            message=_("Nous avons examiné votre candidature pour %(prog)s.") % {
   299→                'prog': candidature.campagne.filiere
   300→            },
   301→            notification_type='candidacy',
   302→            created_by=request.user,
   303→        )
   304→        log_staff_action(request.user, 'candidacy_rejected',
   305→                         f"Candidature {candidature.reference} refusée. Raison: {rejection_reason[:100]}",
   306→                         target_user=candidature.utilisateur)
   307→        messages.success(request, _("Candidature de %(name)s refusée.") % {
   308→            'name': f"{candidature.prenom} {candidature.nom}"})
   309→
   310→    elif action == 'review':
   311→        candidature.statut = 'en_revue'
   312→        candidature.commentaire_admin = notes
   313→        candidature.reviewed_by = request.user
   314→        candidature.save()
   315→        log_staff_action(request.user, 'candidacy_reviewing',
   316→                         f"Candidature {candidature.reference} mise en revue",
   317→                         target_user=candidature.utilisateur)
   318→        messages.info(request, _("Candidature marquée comme en cours de révision."))
   319→
   320→    elif action == 'waitlist':
   321→        candidature.statut = 'liste_attente'
   322→        candidature.commentaire_admin = notes
   323→        candidature.date_decision = timezone.now()
   324→        candidature.reviewed_by = request.user
   325→        candidature.reviewed_at = timezone.now()
   326→        candidature.save()
   327→        notify_user(
   328→            candidature.utilisateur,
   329→            title=_("Liste d'attente"),
   330→            message=_("Vous êtes sur la liste d'attente pour %(prog)s.") % {
   331→                'prog': candidature.campagne.filiere
   332→            },
   333→            notification_type='candidacy',
   334→            created_by=request.user,
   335→        )
   336→        log_staff_action(request.user, 'candidacy_waitlisted',
   337→                         f"Candidature {candidature.reference} mise en liste d'attente",
   338→                         target_user=candidature.utilisateur)
   339→        messages.info(request, _("Candidature mise en liste d'attente."))
   340→
   341→    return redirect('admin_panel:candidatures_list')

   410→    )

   412→    demande = get_object_or_404(DemandeMentorat, id=demande_id)

   417→    mentor_user = demande.mentor.profil.utilisateur
   418→
   419→    if action in ['accept', 'reject'] and demande.statut in ['acceptee', 'refusee', 'terminee']:
   420→        messages.warning(request, _("Cette demande a déjà été traitée et ne peut plus être modifiée."))
   421→        return redirect('admin_panel:mentorat_detail', demande_id=demande_id)
   422→
   423→    if action == 'accept':
   424→        demande.accepter()
   425→        if notes:
   426→            demande.admin_notes = notes
   427→            demande.save(update_fields=['admin_notes'])
   428→        try:
   429→            send_mentorship_acceptance_email(demande)
   430→        except Exception as e:
   431→            logger.error('mentorship accept email failed: %s', e)
   432→        try:
   433→            send_mentorship_confirmation_to_mentor(demande.mentor, demande.mentee)
   434→        except Exception as e:
   435→            logger.error('mentorship mentor confirm email failed: %s', e)
   436→        notify_user(
   437→            mentee_user,
   438→            title=_("Demande de mentorat acceptée"),
   439→            message=_("%(mentor)s va vous accompagner comme mentor. 🤝") % {
   440→                'mentor': mentor_user.get_full_name() or mentor_user.username
   441→            },
   442→            notification_type='mentorship',
   443→            link='/mentorat/tableau-bord-mentee/',
   444→            created_by=request.user,
   445→        )
   446→        notify_user(
   447→            mentor_user,
   448→            title=_("Nouveau mentee assigné"),
   449→            message=_("%(mentee)s vous a été assigné comme mentee.") % {
   450→                'mentee': mentee_user.get_full_name() or mentee_user.username
   451→            },
   452→            notification_type='mentorship',
   453→            link='/mentorat/tableau-bord-mentor/',
   454→            created_by=request.user,
   455→        )
   456→        log_staff_action(request.user, 'mentorship_accepted',
   457→                         f"Mentorat accepté: {mentee_user} → {mentor_user}",
   458→                         target_user=mentee_user)
   459→        messages.success(request, _("✅ Demande de mentorat acceptée."))
   460→
   461→    elif action == 'reject':
   462→        rejection_reason = request.POST.get('rejection_reason', '').strip()
   463→        if not rejection_reason:
   464→            messages.error(request, _("La raison de refus est obligatoire."))
   465→            return redirect('admin_panel:mentorat_detail', demande_id=demande_id)
   466→        demande.refuser()
   467→        demande.admin_notes = f"Refusé par {request.user.username}: {rejection_reason}"

   470→            mentee_user,
   471→            title=_("Demande de mentorat"),

   474→            created_by=request.user,
   475→        )

   477→                         f"Mentorat refusé: {mentee_user} → {mentor_user}. {rejection_reason[:80]}",
   478→                         target_user=mentee_user)
   479→        messages.info(request, _("Demande de mentorat refusée."))

   481→    elif action == 'assign_mentor':

   490→                messages.error(request, _("Mentor introuvable."))
   491→        return redirect('admin_panel:mentorat_detail', demande_id=demande_id)
   492→
   493→    return redirect('admin_panel:mentorat_list')

   568→@staff_only
   569→@require_POST
   570→def user_action(request, user_id):
   571→    target = get_object_or_404(User, id=user_id)
   572→    action = request.POST.get('action')
   573→
   574→    if action == 'toggle_active':
   575→        if target == request.user:
   576→            messages.error(request, _("Vous ne pouvez pas vous désactiver vous-même."))
   577→            return redirect('admin_panel:user_detail', user_id=user_id)
   578→        target.is_active = not target.is_active
   579→        target.save(update_fields=['is_active'])
   580→        if target.is_active:
   581→            log_staff_action(request.user, 'user_activated',
   582→                             f"Utilisateur {target.username} activé", target_user=target)
   583→            messages.success(request, _("Compte activé."))
   584→        else:
   585→            log_staff_action(request.user, 'user_deactivated',
   586→                             f"Utilisateur {target.username} désactivé", target_user=target)
   587→            messages.warning(request, _("Compte désactivé."))
   588→
   589→    elif action == 'toggle_staff':
   590→        if target == request.user:
   591→            messages.error(request, _("Vous ne pouvez pas modifier votre propre statut staff."))
   592→            return redirect('admin_panel:user_detail', user_id=user_id)
   593→        target.is_staff = not target.is_staff
   594→        target.save(update_fields=['is_staff'])
   595→        messages.success(request, _("Statut staff mis à jour."))
   596→
   597→    elif action == 'send_notification':
   598→        title = request.POST.get('notif_title', '').strip()
   599→        msg_text = request.POST.get('notif_message', '').strip()
   600→        if title and msg_text:
   601→            notify_user(target, title=title, message=msg_text,
   602→                        notification_type='info', created_by=request.user)
   603→            log_staff_action(request.user, 'notification_sent',
   604→                             f"Notification envoyée à {target.username}: {title}",
   605→                             target_user=target)
   606→            messages.success(request, _("Notification envoyée."))
   607→        else:
   608→            messages.error(request, _("Titre et message requis."))
   609→        return redirect('admin_panel:user_detail', user_id=user_id)
   610→
   611→    elif action == 'reset_password':
   612→        form = PasswordResetForm({'email': target.email})
   613→        if form.is_valid():
   614→            form.save(
   615→                request=request,
   616→                use_https=request.is_secure(),
   617→                email_template_name='registration/password_reset_email.html',
   618→                subject_template_name='registration/password_reset_subject.txt',
   619→            )
   620→            messages.success(
   621→                request,
   622→                _("Email de réinitialisation envoyé à %(email)s") % {'email': target.email}
   623→            )
   624→        else:
   625→            messages.error(request, _("Impossible d'envoyer l'email (adresse invalide ?)"))
   626→        return redirect('admin_panel:user_detail', user_id=user_id)
   627→
   628→    return redirect('admin_panel:user_detail', user_id=user_id)
