   273→   273→@login_required
   274→   274→def tableau_de_bord_mentor(request):
   275→   275→    """
   276→   276→    Tableau de bord du mentor.
   277→   277→    """
   278→   278→    if not hasattr(request.user, 'profil') or not hasattr(request.user.profil, 'mentorat_mentor'):
   279→   279→        messages.error(request, _("Vous n'êtes pas inscrit en tant que mentor."))
   280→   280→        return redirect('mentorat:devenir_mentor')
   281→   281→
   282→   282→    mentor = request.user.profil.mentorat_mentor
   283→   283→
   284→   284→    demandes_recues = DemandeMentorat.objects.filter(
   285→   285→        mentor=mentor,
   286→   286→        statut='en_attente'
   287→   287→    ).select_related('mentee__profil__utilisateur')
   288→   288→
   289→   289→    relations_actives = RelationMentorat.objects.filter(
   290→   290→        mentor=mentor,
   291→   291→        est_active=True
   292→   292→    ).select_related('mentee__profil__utilisateur')
   293→   293→
   294→   294→    seances_a_venir = SeanceMentorat.objects.filter(
   295→   295→        relation__mentor=mentor,
   296→   296→        date_heure__gte=timezone.now() - timedelta(hours=4),
   297→   297→        statut='planifiee'
   298→   298→    ).select_related('relation__mentee__profil__utilisateur').order_by('date_heure')[:5]
   299→   299→
   300→   300→    # Revenus du mentor
   301→   301→    paiements_confirmes = PaiementSeance.objects.filter(
   302→   302→        seance__relation__mentor=mentor,
   303→   303→        statut='confirme'
   304→   304→    )
   305→   305→    paiements_en_attente = PaiementSeance.objects.filter(
   306→   306→        seance__relation__mentor=mentor,
   307→   307→        statut='preuve_soumise'
   308→   308→    )
   309→   309→    revenus_confirmes = paiements_confirmes.aggregate(total=Sum('montant_mentor'))['total'] or 0
   310→   310→    revenus_en_attente = paiements_en_attente.aggregate(total=Sum('montant_mentor'))['total'] or 0
   311→   311→
   312→   312→    context = {
   313→   313→        'mentor': mentor,
   314→   314→        'demandes_recues': demandes_recues,
   315→   315→        'relations_actives': relations_actives,
   316→   316→        'seances_a_venir': seances_a_venir,
   317→   317→        'revenus_confirmes': revenus_confirmes,
   318→   318→        'revenus_en_attente': revenus_en_attente,
   319→   319→        'nb_paiements_confirmes': paiements_confirmes.count(),
   320→   320→    }
   321→   321→    return render(request, 'mentorat/tableau_de_bord_mentor.html', context)

   361→   361→@staff_member_required
   362→   362→def application_detail(request, application_pk):
   363→   363→    """Détail et modération d'une candidature de mentor."""
   364→   364→    application = get_object_or_404(MentorApplication.objects.select_related('profil__utilisateur'), pk=application_pk)
   365→   365→    if request.method == 'POST':
   366→   366→        action = request.POST.get('action')
   367→   367→        notes = request.POST.get('rejection_notes', '')
   368→   368→        if action in ['approve', 'reject'] and application.status in ['approved', 'rejected']:
   369→   369→            messages.warning(request, _('Cette candidature a déjà été traitée et ne peut plus être modifiée.'))
   370→   370→            return redirect('mentorat:application_detail', application_pk=application.pk)
   371→   371→        if action == 'approve':
   372→   372→            application.approve()
   373→   373→            try:
   374→   374→                send_mentor_application_approval_email(application)
   375→   375→            except Exception as e:
   376→   376→                logger.error('mentor application approval email failed: %s', e)
   377→   377→            notify_user(
   378→   378→                application.profil.utilisateur,
   379→   379→                title=_('Candidature mentor approuvée'),
   380→   380→                message=_('Votre candidature pour devenir mentor a été approuvée.'),
   381→   381→                notification_type='mentorship',
   382→   382→                link=reverse('mentorat:applications_list'),
   383→   383→                created_by=request.user,
   384→   384→            )
   385→   385→            messages.success(request, _('La candidature a été approuvée et le mentor a été activé.'))
   386→   386→        elif action == 'reject':
   387→   387→            application.reject(notes=notes)
   388→   388→            try:
   389→   389→                send_mentor_application_rejection_email(application)
   390→   390→            except Exception as e:
   391→   391→                logger.error('mentor application rejection email failed: %s', e)
   392→   392→            notify_user(
   393→   393→                application.profil.utilisateur,
   394→   394→                title=_('Candidature mentor rejetée'),
   395→   395→                message=_('Votre candidature pour devenir mentor a été refusée.'),
   396→   396→                notification_type='mentorship',
   397→   397→                link=reverse('mentorat:applications_list'),
   398→   398→                created_by=request.user,
   399→   399→            )
   400→   400→            messages.success(request, _('La candidature a été rejetée.'))
   401→   401→        return redirect('mentorat:application_detail', application_pk=application.pk)
   402→   402→    return render(request, 'mentorat/admin_application_detail.html', {
   403→   403→        'application': application,
   404→   404→    })
