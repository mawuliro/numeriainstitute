    25→def extraire_id_youtube(url: str | None):
    26→    """Extract YouTube video ID from common URL formats."""
    27→    if not url or 'vimeo' in (url or '').lower():
    28→        return None
    29→    url = url.strip()
    30→    if 'youtu.be/' in url:
    31→        return url.split('youtu.be/')[-1].split('?')[0].strip('/')
    32→    if '/shorts/' in url:
    33→        return url.split('/shorts/')[-1].split('?')[0].strip('/')
    34→    if '/embed/' in url:
    35→        return url.split('/embed/')[-1].split('?')[0].strip('/')
    36→    if 'v=' in url:
    37→        return url.split('v=')[-1].split('&')[0].strip()
    38→    return None
    39→
    40→
    41→def convertir_url_youtube(url: str | None):
    42→    """Convert YouTube/Vimeo URL to embed URL (display-time only)."""
    43→    if not url:
    44→        return url
    45→
    46→    url = url.strip()
    47→    u = url.lower()
    48→
    49→    # Vimeo
    50→    if 'vimeo.com' in u:
    51→        if '/video/' in u:
    52→            video_id = url.split('/video/')[-1].split('?')[0].split('/')[0]
    53→        else:
    54→            video_id = url.split('vimeo.com/')[-1].split('?')[0].split('/')[0]
    55→        if video_id.isdigit():
    56→            return f'https://player.vimeo.com/video/{video_id}?color=E8A020&title=0&byline=0'
    57→        return url
    58→
    59→    # YouTube
    60→    if 'youtube-nocookie.com/embed/' in u:
    61→        return url
    62→    if 'youtube.com/embed/' in u:
    63→        video_id = url.split('/embed/')[-1].split('?')[0].strip('/')
    64→        return f'https://www.youtube-nocookie.com/embed/{video_id}?rel=0&modestbranding=1'
    65→    if 'youtu.be/' in u:
    66→        video_id = url.split('youtu.be/')[-1].split('?')[0].strip('/')
    67→        return f'https://www.youtube-nocookie.com/embed/{video_id}?rel=0&modestbranding=1'
    68→    if '/shorts/' in u:
    69→        video_id = url.split('/shorts/')[-1].split('?')[0].strip('/')
    70→        return f'https://www.youtube-nocookie.com/embed/{video_id}?rel=0&modestbranding=1'
    71→    if 'youtube.com' in u and 'v=' in url:
    72→        video_id = url.split('v=')[-1].split('&')[0].strip()
    73→        return f'https://www.youtube-nocookie.com/embed/{video_id}?rel=0&modestbranding=1'
    74→    return url

   134→        return self.title
   135→
   136→    def save(self, *args, **kwargs):
   137→        if not self.slug:
   138→            base = slugify(self.title)
   139→            slug = base
   140→            n = 1
   141→            while Course.objects.filter(slug=slug).exclude(pk=self.pk).exists():
   142→                slug = f'{base}-{n}'
   143→                n += 1
   144→            self.slug = slug
   145→        super().save(*args, **kwargs)

   259→        """Return a safe filename for the PDF certificate download."""

   262→        return f'certificat-{nom}-{cours}.pdf'

   389→    def get_payload(self, user=None, preview=False):
   390→        """Build a sanitised dict for the lesson_blocks_render.html template.
   391→
   392→        Never includes solution_code or test_code in plain text —
   393→        test_code is base64-encoded for client-side evaluation only.
   394→        """
   395→        import base64
   396→
   397→        data = {
   398→            'id': self.id,
   399→            'type': self.block_type,
   400→            'order': self.order,
   401→        }
   402→
   403→        if self.block_type == 'text':
   404→            data['text_content'] = self.text_content or ''
   405→
   406→        elif self.block_type == 'video':
   407→            data['embed_url'] = self._embed_url(self.video_url)
   408→            data['video_caption'] = self.video_caption or ''
   409→
   410→        elif self.block_type == 'sandbox':
   411→            data['title'] = self.sandbox_title or 'Essaie toi-même'
   412→            data['initial_code'] = self.sandbox_initial_code or ''
   413→
   414→        elif self.block_type == 'code_exercise':
   415→            ex = self.code_exercise
   416→            if ex:
   417→                data['exercise_id'] = ex.id
   418→                data['id'] = ex.id
   419→                data['title'] = ex.title
   420→                data['instructions'] = ex.instructions or ''
   421→                data['difficulty'] = ex.difficulty
   422→                data['points'] = ex.points
   423→                data['hint'] = ex.hint or ''
   424→                data['max_attempts'] = ex.max_attempts
   425→                data['starter_code'] = ex.starter_code or ''
   426→                data['expected_output'] = ex.expected_output or ''
   427→                data['evaluation_mode'] = ex.evaluation_mode
   428→                data['test_code_b64'] = base64.b64encode(
   429→                    (ex.test_code or '').encode()
   430→                ).decode() if ex.test_code else ''
   431→                data['order'] = ex.order
   432→                data['is_solved'] = False
   433→                data['points_earned'] = 0
   434→                data['attempts_used'] = 0
   435→                if user and not preview:
   436→                    from cours.models import StudentProgress, ExerciseAttempt
   437→                    progress = StudentProgress.objects.filter(
   438→                        student=user, exercise_type='code',
   439→                        exercise_id=ex.id
   440→                    ).first()
   441→                    if progress:
   442→                        data['is_solved'] = progress.is_solved
   443→                        data['points_earned'] = progress.points_earned
   444→                    data['attempts_used'] = ExerciseAttempt.objects.filter(
   445→                        exercise_type='code', exercise_id=ex.id, student=user
   446→                    ).count()
   447→
   448→        elif self.block_type == 'mcq':

   450→            if ex:
   451→                data['mcq_id'] = ex.id
   452→                data['mcq_title'] = ex.title
   453→                data['title'] = ex.title
   454→                data['question'] = ex.question or ''
   455→                data['hint'] = ex.hint or ''
   456→                data['allow_multiple'] = ex.allow_multiple_correct
   457→                data['shuffle'] = ex.shuffle_choices
   458→                data['max_attempts'] = ex.max_attempts
   459→                data['points'] = ex.points
   460→                data['difficulty'] = ex.difficulty
   461→                data['explanation'] = ''  # hidden until solved/exhausted
   462→                data['is_solved'] = False
   463→                data['points_earned'] = 0
   464→                data['attempts_used'] = 0
   465→                data['correct_ids'] = []
   466→                data['choices'] = [
   467→                    {'id': c.id, 'text': c.text, 'order': c.order}
   468→                    for c in ex.choices.all().order_by('order')
   469→                ]
   470→                if user and not preview:
   471→                    from cours.models import StudentProgress, ExerciseAttempt
   472→                    progress = StudentProgress.objects.filter(
   473→                        student=user, exercise_type='mcq',
   474→                        exercise_id=ex.id
   475→                    ).first()
   476→                    if progress:

   504→                        student=user, exercise_type='fill_blank',
   505→                        exercise_id=ex.id
   506→                    ).first()
   507→                    if progress:
   508→                        data['is_solved'] = progress.is_solved
   509→                    data['attempts_used'] = ExerciseAttempt.objects.filter(
   510→                        exercise_type='fill_blank', exercise_id=ex.id, student=user
   511→                    ).count()
   512→
   513→        elif self.block_type == 'true_false':
   514→            ex = self.true_false
   515→            if ex:
   516→                data['true_false_id'] = ex.id
   517→                data['title'] = ex.title
   518→                data['statements'] = ex.statements or []
   519→                data['points_per_statement'] = ex.points_per_statement

   535→                        exercise_type='true_false', exercise_id=ex.id, student=user

   537→
   538→        elif self.block_type == 'code_order':
   539→            ex = self.code_order
   540→            if ex:
   541→                data['code_order_id'] = ex.id
   542→                data['title'] = ex.title

   559→                data['points'] = ex.points
   560→                data['difficulty'] = ex.difficulty
   561→                data['hint'] = ex.hint or ''
   562→                data['is_solved'] = False
   563→                data['attempts_used'] = 0
   564→
   565→        elif self.block_type == 'short_answer':
   566→            ex = self.short_answer
   567→            if ex:
   568→                data['short_answer_id'] = ex.id
   569→                data['title'] = ex.title
   570→                data['instructions'] = ex.instructions or ''
   571→                data['points'] = ex.points
   572→                data['difficulty'] = ex.difficulty
   573→                data['hint'] = ex.hint or ''
   574→                data['max_attempts'] = ex.max_attempts
   575→                data['is_solved'] = False
   576→                data['attempts_used'] = 0
   577→
   578→        elif self.block_type == 'interactive_lab':
   579→            ex = self.interactive_lab
   580→            if ex:
   581→                data['lab_id'] = ex.id
   582→                data['title'] = ex.title
   583→                data['instructions'] = ex.instructions or ''
   584→                data['simulation_code'] = ex.simulation_code
   585→                data['slider_config'] = ex.slider_config
   586→                data['challenges'] = ex.challenges
   587→                data['points'] = ex.points
   588→                data['difficulty'] = ex.difficulty
   589→                # Student progress (if any)
   590→                data['current_challenge_id'] = ''
   591→                data['challenges_solved'] = []
   592→                data['is_completed'] = False
   593→                if user and not preview:
   594→                    from cours.models import LabProgress
   595→                    p = LabProgress.objects.filter(student=user, lab=ex).first()
   596→                    if p:
   597→                        data['current_challenge_id'] = p.current_challenge_id
   598→                        data['challenges_solved'] = p.challenges_solved
   599→                        data['is_completed'] = p.is_completed
   600→
   601→        return data
   602→
   603→    @staticmethod
   604→    def _embed_url(url):
   605→        """Convert YouTube/Vimeo URLs to embeddable iframe URLs."""
   606→        if not url:
   607→            return ''
   608→        u = url.strip()
   609→        if 'youtube.com/watch' in u and 'v=' in u:
   610→            vid = u.split('v=', 1)[1].split('&', 1)[0]
   611→            return f'https://www.youtube.com/embed/{vid}'
   612→        if 'youtu.be/' in u:
   613→            vid = u.split('youtu.be/', 1)[1].split('?', 1)[0]
   614→            return f'https://www.youtube.com/embed/{vid}'
   615→        if 'vimeo.com/' in u and '/embed/' not in u:
   616→            vid = u.rstrip('/').split('/')[-1]
   617→            return f'https://player.vimeo.com/video/{vid}'
   618→        return u
