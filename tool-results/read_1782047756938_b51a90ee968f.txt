    20→        if options["clean"]:
    21→            d, _ = Course.objects.filter(slug=slug).delete()

    27→            "category": "physique", "level": "debutant", "language": "fr",
    28→            "price": 0, "is_free": True, "status": status, "estimated_hours": 40})
    29→        if not created: self.stdout.write(self.style.WARNING("Course exists - updating."))
    30→        for md in COURSE_STRUCTURE:
    31→            mod = self._mod(course, md)
    32→            for ld in md["lessons"]:
    33→                les = self._les(course, mod, ld)
    34→                self._blocks(les, ld["blocks"])
    35→        self.stdout.write(self.style.SUCCESS(
    36→            f"\n✓ Seeded: {course.title}\n  Modules: {CourseModule.objects.filter(course=course).count()}\n"
    37→            f"  Lessons: {CourseLesson.objects.filter(course=course).count()}\n"
    38→            f"  Blocks: {LessonBlock.objects.filter(course_lesson__course=course).count()}\n"))
    39→    def _mod(self, course, d):
    40→        m, _ = CourseModule.objects.get_or_create(course=course, title=d["title"],
    41→            defaults={"description": d.get("description",""), "order": d["order"], "is_active": True})
    42→        return m
    43→    def _les(self, course, mod, d):
    44→        s = d.get("slug") or d["title"].lower().replace(" ","-").replace("'","-")
    45→        l, _ = CourseLesson.objects.get_or_create(course=course, module=mod, title=d["title"],
    46→            defaults={"slug": s, "order": d["order"], "estimated_minutes": d.get("minutes",25),
    47→                      "is_free_preview": True, "is_active": True})
    48→        return l
    49→    def _blocks(self, lesson, blocks):
    50→        LessonBlock.objects.filter(course_lesson=lesson).delete()
