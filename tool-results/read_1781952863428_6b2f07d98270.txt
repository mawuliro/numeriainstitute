    72→@login_required
    73→def devenir_mentor(request):
    74→    """
    75→    Candidature pour devenir mentor.
    76→    """
    77→    if not hasattr(request.user, 'profil'):
    78→        messages.error(request, _("Votre profil n'est pas configuré. Veuillez contacter l'administrateur."))
    79→        return redirect('mentorat:index')
    80→
    81→    profil = request.user.profil
    82→    # Si le mentor est déjà approuvé, on lui permet d'éditer son profil.
    83→    if hasattr(profil, 'mentorat_mentor'):
    84→        mentor = profil.mentorat_mentor
    85→        if request.method == 'POST':
    86→            form = MentorProfileForm(request.POST, request.FILES, instance=mentor, profil=profil)
    87→            if form.is_valid():
    88→                form.save()
    89→                messages.success(request, _("Votre profil mentor a été mis à jour."))
    90→                return redirect('mentorat:tableau_de_bord_mentor')
    91→        else:
    92→            form = MentorProfileForm(instance=mentor, profil=profil)
    93→
    94→        return render(request, 'mentorat/devenir_mentor.html', {
    95→            'form': form,
    96→            'is_mentor': True,
    97→            'mentor': mentor,
    98→        })
    99→
   100→    application = getattr(profil, 'application_mentor', None)
   101→    if application and application.status == 'pending':
   102→        return render(request, 'mentorat/devenir_mentor.html', {
   103→            'application': application,
   104→            'application_pending': True,
   105→        })
   106→
   107→    if request.method == 'POST':
   108→        form = MentorApplicationForm(request.POST, request.FILES, instance=application if application and application.status == 'rejected' else None)
   109→        if form.is_valid():
   110→            application = form.save(commit=False)
   111→            application.profil = profil
   112→            application.status = 'pending'
   113→            application.save()
   114→            if form.cleaned_data.get('certificate'):
   115→                application.certificate = form.cleaned_data['certificate']
   116→            if form.cleaned_data.get('cv'):
   117→                application.cv = form.cleaned_data['cv']
   118→            application.save()
   119→            messages.success(request, _("Votre candidature a bien été envoyée. Un administrateur la validera prochainement."))
   120→            return redirect('mentorat:devenir_mentor')
   121→    else:
   122→        form = MentorApplicationForm(instance=application if application and application.status == 'rejected' else None)
   123→
   124→    return render(request, 'mentorat/devenir_mentor.html', {
   125→        'form': form,
   126→        'application_rejected': application and application.status == 'rejected',
   127→        'rejection_notes': application.rejection_notes if application and application.status == 'rejected' else '',
   128→    })

   162→    """
   163→    Liste des mentors disponibles.
   164→    """
   165→    mentors = Mentor.objects.filter(est_actif=True, profil__est_menteur=True).select_related('profil__utilisateur')
   166→
   167→    # Filtres
   168→    domaine = request.GET.get('domaine')
   169→    if domaine:
   170→        mentors = mentors.filter(domaines_expertise=domaine)
   171→
   172→    niveau = request.GET.get('niveau')

   174→        mentors = mentors.filter(niveau_experience=niveau)
   175→
   176→    # Recherche
   177→    query = request.GET.get('q')
   178→    if query:
   179→        mentors = mentors.filter(
   180→            Q(profil__utilisateur__first_name__icontains=query) |
   181→            Q(profil__utilisateur__last_name__icontains=query) |
   182→            Q(bio_mentorat__icontains=query)
   183→        )
   184→
   185→    context = {
   186→        'mentors': mentors,
   187→        'domaines': Mentor.DOMAINES,
   188→        'niveaux': Mentor.NIVEAUX_EXPERIENCE,
   189→    }
   190→    return render(request, 'mentorat/liste_mentors.html', context)

   224→@login_required
   225→def demander_mentorat(request, mentor_pk):
   226→    """
   227→    Faire une demande de mentorat.
   228→    """
   229→    mentor = get_object_or_404(Mentor, pk=mentor_pk)
   230→
   231→    # Vérifications
   232→    if not hasattr(request.user, 'profil') or not hasattr(request.user.profil, 'mentorat_mentee'):
   233→        messages.error(request, _("Vous devez d'abord vous inscrire en tant que mentoré."))
   234→        return redirect('mentorat:devenir_mentee')
   235→
   236→    mentee = request.user.profil.mentorat_mentee
   237→
   238→    # Vérifier qu'il n'y a pas déjà une demande ou relation
   239→    if DemandeMentorat.objects.filter(
   240→        mentee=mentee,
   241→        mentor=mentor,
   242→        statut__in=['en_attente', 'acceptee']
   243→    ).exists():
   244→        messages.warning(request, _("Vous avez déjà une demande en cours avec ce mentor."))
   245→        return redirect('mentorat:detail_mentor', pk=mentor_pk)
   246→
   247→    if RelationMentorat.objects.filter(
   248→        mentee=mentee,
   249→        mentor=mentor,
   250→        est_active=True
   251→    ).exists():
   252→        messages.info(request, _("Vous êtes déjà en relation de mentorat avec ce mentor."))
   253→        return redirect('mentorat:tableau_de_bord_mentee')
   254→
   255→    if request.method == 'POST':
   256→        form = DemandeMentoratForm(request.POST)
   257→        if form.is_valid():
   258→            demande = form.save(commit=False)
   259→            demande.mentee = mentee
   260→            demande.mentor = mentor
   261→            demande.save()
   262→            messages.success(request, _("Votre demande de mentorat a été envoyée !"))
   263→            return redirect('mentorat:tableau_de_bord_mentee')
   264→    else:
   265→        form = DemandeMentoratForm()
   266→
   267→    return render(request, 'mentorat/demander_mentorat.html', {
   268→        'form': form,
   269→        'mentor': mentor
   270→    })

   273→@login_required
   274→def tableau_de_bord_mentor(request):
   275→    """
   276→    Tableau de bord du mentor.
   277→    """
   278→    if not hasattr(request.user, 'profil') or not hasattr(request.user.profil, 'mentorat_mentor'):
   279→        messages.error(request, _("Vous n'êtes pas inscrit en tant que mentor."))
   280→        return redirect('mentorat:devenir_mentor')
   281→
   282→    mentor = request.user.profil.mentorat_mentor
   283→
   284→    demandes_recues = DemandeMentorat.objects.filter(
   285→        mentor=mentor,
   286→        statut='en_attente'
   287→    ).select_related('mentee__profil__utilisateur')
   288→
   289→    relations_actives = RelationMentorat.objects.filter(
   290→        mentor=mentor,
   291→        est_active=True
   292→    ).select_related('mentee__profil__utilisateur')
   293→
   294→    seances_a_venir = SeanceMentorat.objects.filter(
   295→        relation__mentor=mentor,
   296→        date_heure__gte=timezone.now() - timedelta(hours=4),
   297→        statut='planifiee'
   298→    ).select_related('relation__mentee__profil__utilisateur').order_by('date_heure')[:5]
   299→
   300→    # Revenus du mentor
   301→    paiements_confirmes = PaiementSeance.objects.filter(
   302→        seance__relation__mentor=mentor,
   303→        statut='confirme'
   304→    )
   305→    paiements_en_attente = PaiementSeance.objects.filter(
   306→        seance__relation__mentor=mentor,
   307→        statut='preuve_soumise'
   308→    )
   309→    revenus_confirmes = paiements_confirmes.aggregate(total=Sum('montant_mentor'))['total'] or 0
   310→    revenus_en_attente = paiements_en_attente.aggregate(total=Sum('montant_mentor'))['total'] or 0
   311→
   312→    context = {
   313→        'mentor': mentor,
   314→        'demandes_recues': demandes_recues,
   315→        'relations_actives': relations_actives,
   316→        'seances_a_venir': seances_a_venir,
   317→        'revenus_confirmes': revenus_confirmes,
   318→        'revenus_en_attente': revenus_en_attente,
   319→        'nb_paiements_confirmes': paiements_confirmes.count(),
   320→    }
   321→    return render(request, 'mentorat/tableau_de_bord_mentor.html', context)

   361→@staff_member_required
   362→def application_detail(request, application_pk):
   363→    """Détail et modération d'une candidature de mentor."""
   364→    application = get_object_or_404(MentorApplication.objects.select_related('profil__utilisateur'), pk=application_pk)
   365→    if request.method == 'POST':
   366→        action = request.POST.get('action')
   367→        notes = request.POST.get('rejection_notes', '')
   368→        if action in ['approve', 'reject'] and application.status in ['approved', 'rejected']:
   369→            messages.warning(request, _('Cette candidature a déjà été traitée et ne peut plus être modifiée.'))
   370→            return redirect('mentorat:application_detail', application_pk=application.pk)
   371→        if action == 'approve':
   372→            application.approve()
   373→            try:
   374→                send_mentor_application_approval_email(application)
   375→            except Exception as e:
   376→                logger.error('mentor application approval email failed: %s', e)
   377→            notify_user(
   378→                application.profil.utilisateur,
   379→                title=_('Candidature mentor approuvée'),
   380→                message=_('Votre candidature pour devenir mentor a été approuvée.'),
   381→                notification_type='mentorship',
   382→                link=reverse('mentorat:applications_list'),
   383→                created_by=request.user,
   384→            )
   385→            messages.success(request, _('La candidature a été approuvée et le mentor a été activé.'))
   386→        elif action == 'reject':
   387→            application.reject(notes=notes)
   388→            try:
   389→                send_mentor_application_rejection_email(application)
   390→            except Exception as e:
   391→                logger.error('mentor application rejection email failed: %s', e)
   392→            notify_user(
   393→                application.profil.utilisateur,
   394→                title=_('Candidature mentor rejetée'),
   395→                message=_('Votre candidature pour devenir mentor a été refusée.'),
   396→                notification_type='mentorship',
   397→                link=reverse('mentorat:applications_list'),
   398→                created_by=request.user,
   399→            )
   400→            messages.success(request, _('La candidature a été rejetée.'))
   401→        return redirect('mentorat:application_detail', application_pk=application.pk)
   402→    return render(request, 'mentorat/admin_application_detail.html', {
   403→        'application': application,
   404→    })

   407→@login_required
   408→def video_seance(request, seance_pk):
   409→    """Page de visioconférence pour une séance de mentorat en visio."""
   410→    seance = get_object_or_404(
   411→        SeanceMentorat.objects.select_related(
   412→            'relation__mentor__profil__utilisateur',
   413→            'relation__mentee__profil__utilisateur',
   414→            'relation__mentor__profil',
   415→            'relation__mentee__profil'
   416→        ),
   417→        pk=seance_pk,
   418→        statut__in=['planifiee', 'en_cours']
   419→    )
   420→
   421→    if seance.modalite != 'visio':
   422→        raise Http404("Cette séance n'est pas prévue en visioconférence.")
   423→
   424→    if not _is_seance_participant(request.user, seance):
   425→        raise Http404("Accès refusé.")
   426→
   427→    mentor_user = seance.relation.mentor.profil.utilisateur
   428→    mentee_user = seance.relation.mentee.profil.utilisateur
   429→    current_user_id = request.user.id
   430→    if current_user_id == mentor_user.id:
   431→        role = 'mentor'
   432→    elif current_user_id == mentee_user.id:
   433→        role = 'mentee'
   434→    else:
   435→        raise Http404("Accès refusé.")
   436→
   437→    participants = [
   438→        {
   439→            'id': mentor_user.id,
   440→            'name': mentor_user.get_full_name() or mentor_user.username,
   441→            'role': 'mentor',
   442→        },
   443→        {
   444→            'id': mentee_user.id,
   445→            'name': mentee_user.get_full_name() or mentee_user.username,
   446→            'role': 'mentee',
   447→        },
   448→    ]
   449→
   450→    context = {
   451→        'room_type': 'mentorat_seance',
   452→        'room_pk': seance.pk,
   453→        'room_title': seance.titre,
   454→        'room_description': seance.description or seance.relation.mentor.profil.utilisateur.get_full_name() or seance.relation.mentee.profil.utilisateur.get_full_name() or seance.titre,
   455→        'room_modalite': seance.get_modalite_display(),
   456→        'room_date': seance.date_heure.strftime('%d %B %Y'),
   457→        'room_time': seance.date_heure.strftime('%H:%M'),
   458→        'room_duration': f"{seance.duree_minutes} minutes",
   459→        'room_participants': participants,
   460→        'participants_count': len(participants),
   461→        'participant_label': 'Mentor et mentee',
   462→        'current_user_id': current_user_id,
   463→        'current_user_name': request.user.get_full_name() or request.user.username,
   464→        'role_label': 'Mentor' if role == 'mentor' else 'Mentee',
   465→        'back_url': reverse('mentorat:detail_relation', kwargs={'pk': seance.relation.pk}),
   466→        'back_label': "Retour à la relation",
   467→        'room_notes': "Activez votre micro et votre caméra, puis lancez la session en groupe.",
   468→    }
   469→    return render(request, 'video_room.html', context)

   516→        messages.error(request, _("Accès non autorisé."))

   518→
   519→    if demande.mentor != request.user.profil.mentorat_mentor:

   525→        try:
   526→            send_mentorship_acceptance_email(demande, language=getattr(request, 'LANGUAGE_CODE', None))

   528→            pass
   529→        try:
   530→            from notifications.notifications import notify_user
   531→            mentor_user = demande.mentor.profil.utilisateur

   534→                mentee_user,

   540→        except Exception:
   541→            pass
   542→        messages.success(request, _("Demande de %(mentee)s acceptée !") % {'mentee': demande.mentee})
   543→    elif action == 'refuser':
   544→        demande.refuser()
   545→        messages.success(request, _("Demande de %(mentee)s refusée.") % {'mentee': demande.mentee})

   689→@login_required
   690→def paiement_seance(request, seance_pk):
   691→    """
   692→    Page de paiement pour une séance payante.
   693→    Accessible uniquement par le mentee de la relation.
   694→    """
   695→    seance = get_object_or_404(SeanceMentorat, pk=seance_pk)
   696→
   697→    # Seul le mentee doit payer
   698→    user_profil = request.user.profil
   699→    if not (hasattr(user_profil, 'mentorat_mentee') and seance.relation.mentee == user_profil.mentorat_mentee):
   700→        messages.error(request, _("Accès non autorisé."))
   701→        return redirect('mentorat:index')
   702→
   703→    # Récupérer ou créer le paiement de séance associé
   704→    paiement_seance, _ = PaiementSeance.objects.get_or_create(
   705→        seance=seance,
   706→        defaults={
   707→            'montant_total': seance.relation.mentor.tarif_par_seance,
   708→            'commission_numeria': round(seance.relation.mentor.tarif_par_seance * PaiementSeance.TAUX_COMMISSION / 100),
   709→            'montant_mentor': round(seance.relation.mentor.tarif_par_seance * (100 - PaiementSeance.TAUX_COMMISSION) / 100),
   710→        }
   711→    )
   712→
   713→    if paiement_seance.statut == 'confirme':
   714→        messages.info(request, _("Ce paiement est déjà confirmé."))
   715→        return redirect('mentorat:detail_relation', pk=seance.relation.pk)
   716→
   717→    providers_disponibles = PAYMENT_PROVIDERS
   718→
   719→    if request.method == 'POST' and 'provider' in request.POST:
   720→        provider = request.POST.get('provider', 'sandbox')
   721→        try:
   722→            paiement, nouveau = creer_paiement(
   723→                etudiant=request.user,
   724→                paiement_seance=paiement_seance,
   725→                provider=provider
   726→            )
   727→
   728→            paiement_seance.lier_paiement(paiement)
   729→
   730→            if paiement.statut != 'reussi':
   731→                traiter_paiement(paiement, provider)
   732→
   733→            paiement_seance.confirmer()
   734→
   735→            messages.success(
   736→                request,
   737→                "✅ Paiement réussi ! Votre séance de mentorat est confirmée."
   738→            )
   739→            return redirect('mentorat:detail_relation', pk=seance.relation.pk)
   740→
   741→        except NotImplementedError:
   742→            messages.warning(
   743→                request,
   744→                f"⚠️ Le provider '{provider}' n'est pas encore disponible. Contactez-nous à {settings.CONTACT_EMAIL}"
   745→            )
   746→            return redirect('mentorat:paiement_seance', seance_pk=seance.pk)
   747→
   748→        except Exception as e:
   749→            messages.error(request, f"❌ Erreur lors du paiement : {str(e)}")
   750→            return redirect('mentorat:paiement_seance', seance_pk=seance.pk)
   751→
   752→    if request.method == 'POST':
   753→        form = PaiementSeanceForm(request.POST, request.FILES, instance=paiement_seance)
   754→        if form.is_valid():
   755→            p = form.save(commit=False)
   756→            p.statut = 'preuve_soumise'
   757→            p.date_submission = timezone.now()
   758→            p.save()
   759→            messages.success(request, _("Preuve de paiement envoyée. Votre séance sera confirmée après vérification."))
   760→            return redirect('mentorat:detail_relation', pk=seance.relation.pk)
   761→    else:
   762→        form = PaiementSeanceForm(instance=paiement_seance)
   763→
   764→    numeros_paiement = settings.MENTORAT_NUMEROS_PAIEMENT
   765→
   766→    return render(request, 'mentorat/paiement_seance.html', {
   767→        'seance': seance,
   768→        'paiement': paiement_seance,
   769→        'form': form,
   770→        'numeros_paiement': numeros_paiement,
   771→        'providers_disponibles': providers_disponibles,
   772→    })
