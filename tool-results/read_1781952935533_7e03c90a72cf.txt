   224→@login_required
   225→def terminer_lecon(request, lecon_id):
   226→    """Marquer une leçon comme terminée."""
   227→    if request.method != 'POST':
   228→        return redirect('cours:catalogue')
   229→
   230→    lecon = get_object_or_404(CourseLesson, id=lecon_id)
   231→    cours = lecon.course
   232→
   233→    inscription = InscriptionCours.objects.filter(
   234→        etudiant=request.user, course=cours
   235→    ).first()
   236→
   237→    if not inscription:
   238→        messages.error(request, _("Tu n'es pas inscrit à ce cours."))
   239→        return redirect('cours:detail', cours_id=cours.id)
   240→
   241→    progression_lecon, cree = ProgressionLecon.objects.get_or_create(
   242→        etudiant=request.user, course_lesson=lecon
   243→    )
   244→
   245→    if cree:
   246→        total_lecons     = cours.lessons.filter(is_active=True).count()
   247→        lecons_terminees = ProgressionLecon.objects.filter(
   248→            etudiant=request.user, course_lesson__course=cours
   249→        ).count()
   250→
   251→        if total_lecons > 0:
   252→            nouveau_pourcentage     = int((lecons_terminees / total_lecons) * 100)
   253→            inscription.progression = nouveau_pourcentage
   254→
   255→            if lecons_terminees >= total_lecons:
   256→                inscription.est_termine = True
   257→                inscription.date_fin    = timezone.now()
   258→                messages.success(
   259→                    request,
   260→                    _("🎉 Félicitations ! Tu as terminé le cours « %(titre)s » !") % {'titre': cours.title}
   261→                )
   262→            else:
   263→                messages.success(
   264→                    request,
   265→                    _("✅ Leçon terminée ! Progression : %(pct)s%%") % {'pct': nouveau_pourcentage}
   266→                )
   267→            inscription.save()
   268→    else:
   269→        messages.info(request, _("Cette leçon était déjà marquée comme terminée."))
   270→
   271→    # Rediriger vers la leçon suivante si elle existe
   272→    lecon_suivante = cours.lessons.filter(
   273→        is_active=True, order__gt=lecon.order
   274→    ).first()
   275→
   276→    url_detail = reverse('cours:detail', kwargs={'cours_id': cours.id})
   277→    if lecon_suivante:
   278→        return redirect(f'{url_detail}?lecon={lecon_suivante.id}')
   279→    return redirect(url_detail)

   334→    """
   335→    Génère et télécharge le certificat PDF d'un cours terminé.
   336→    Réservé aux cours payants uniquement.
   337→    """
   338→    from .models import Certificat
   339→    from .utils.certificat import generer_certificat_pdf
   340→
   341→    inscription = get_object_or_404(
   342→        InscriptionCours,
   343→        id=inscription_id,
   344→        etudiant=request.user,
   345→        est_termine=True
   346→    )
   347→
   348→    # Vérifier que le cours est payant
   349→    if inscription.course.is_free:
   350→        messages.error(
   351→            request,
   352→            _("Les certificats sont réservés aux cours payants.")
   353→        )
   354→        return redirect('comptes:tableau_de_bord')
   355→
   356→    # Récupérer ou créer le certificat
   357→    certificat, cree = Certificat.objects.get_or_create(
   358→        inscription=inscription,
   359→        defaults={
   360→            'code_verification': uuid.uuid4().hex
   361→        }
   362→    )
   363→
   364→    # Construire l'URL de vérification
   365→    url_verification = request.build_absolute_uri(
   366→        reverse('cours:verifier_certificat',
   367→                kwargs={'code': certificat.code_verification})
   368→    )
   369→
   370→    # Générer le PDF
   371→    pdf = generer_certificat_pdf(inscription, url_verification)
   372→
   373→    # Retourner le PDF en téléchargement
   374→    response = HttpResponse(pdf, content_type='application/pdf')
   375→    response['Content-Disposition'] = (
   376→        f'attachment; filename="{certificat.get_nom_fichier()}"'
   377→    )
   378→    return response

   418→@login_required
   419→@require_POST
   420→def submit_code_exercise(request, exercise_id):
   421→    """
   422→    Receive a Pyodide code submission (client-side evaluation).
   423→    Awards points via ExerciseGrade on first correct solve.
   424→    """
   425→    from .models import CodeExercise, StudentProgress, ExerciseAttempt
   426→    from .grades import notify_exercise_solved
   427→    from .progress import record_submission
   428→    from django.utils import timezone as tz
   429→
   430→    exercise = get_object_or_404(CodeExercise, id=exercise_id, is_active=True)
   431→
   432→    try:
   433→        body = json.loads(request.body)
   434→    except (json.JSONDecodeError, ValueError):
   435→        return JsonResponse({'error': 'Invalid JSON'}, status=400)
   436→
   437→    code        = body.get('code', '')
   438→    output      = body.get('output', '')
   439→    is_correct  = bool(body.get('is_correct', False))
   440→    attempt_num = int(body.get('attempt_number', 1))
   441→    time_spent  = int(body.get('time_spent', 0))
   442→
   443→    # Use StudentProgress (replaces ExerciseGrade) + ExerciseAttempt (replaces StudentCodeSubmission)
   444→    progress, _ = StudentProgress.objects.get_or_create(
   445→        student=request.user,
   446→        exercise_type='code',
   447→        exercise_id=exercise.pk,
   448→        defaults={'attempts': 0},
   449→    )
   450→    already_solved = progress.is_solved
   451→    progress.attempts = attempt_num
   452→
   453→    points_earned = 0
   454→    if is_correct and not already_solved:
   455→        progress.is_solved     = True
   456→        progress.points_earned = exercise.points
   457→        progress.solved_at     = tz.now()
   458→        points_earned          = exercise.points
   459→        notify_exercise_solved(request.user, exercise.title, exercise.points)
   460→    progress.save()
   461→
   462→    ExerciseAttempt.objects.create(
   463→        student=request.user,
   464→        exercise_type='code',
   465→        exercise_id=exercise.pk,
   466→        attempt_number=attempt_num,
   467→        is_correct=is_correct,
   468→        points_earned=points_earned,
   469→        answer_data={
   470→            'code': code,
   471→            'output': output,
   472→            'time_spent_seconds': time_spent,
   473→        },
   474→    )
   475→
   476→    return JsonResponse({
   477→        'success': True,
   478→        'points_earned': points_earned,
   479→        'total_points': points_earned,
   480→        'is_correct': is_correct,
   481→        'already_solved': already_solved,
   482→        'attempts_used': attempt_num,
   483→        'max_attempts': exercise.max_attempts,
   484→    })

   507→@login_required
   508→@require_POST
   509→def submit_mcq(request, mcq_id):
   510→    """
   511→    Receive a student's MCQ answer.
   512→    Never sends correct_choice_ids or explanation before solved/max attempts.
   513→    """
   514→    from .models import MCQExercise, MCQChoice, StudentProgress, ExerciseAttempt
   515→    from .progress import record_submission
   516→    from django.utils import timezone as tz
   517→
   518→    mcq = get_object_or_404(MCQExercise, id=mcq_id, is_active=True)
   519→
   520→    try:
   521→        body = json.loads(request.body)
   522→        selected_ids = [int(x) for x in body.get('selected_choice_ids', [])]
   523→    except (json.JSONDecodeError, ValueError, TypeError):
   524→        return JsonResponse({'error': 'Invalid JSON'}, status=400)
   525→
   526→    # Security: verify all selected choices belong to this exercise
   527→    selected_choices = list(MCQChoice.objects.filter(id__in=selected_ids, exercise=mcq))
   528→    if len(selected_choices) != len(selected_ids):
   529→        return JsonResponse({'error': 'Invalid choice IDs'}, status=400)
   530→
   531→    # Use StudentProgress instead of removed MCQGrade
   532→    progress, _ = StudentProgress.objects.get_or_create(
   533→        student=request.user,
   534→        exercise_type='mcq',
   535→        exercise_id=mcq.pk,
   536→        defaults={'attempts': 0},
   537→    )
   538→
   539→    if progress.is_solved:
   540→        return JsonResponse({
   541→            'is_correct': True,
   542→            'already_solved': True,
   543→            'points_earned': progress.points_earned,
   544→            'attempts_used': progress.attempts,
   545→            'max_attempts': mcq.max_attempts,
   546→        })
   547→
   548→    # Check max attempts
   549→    if mcq.max_attempts > 0 and progress.attempts >= mcq.max_attempts:
   550→        correct_ids = list(MCQChoice.objects.filter(exercise=mcq, is_correct=True).values_list('id', flat=True))
   551→        return JsonResponse({
   552→            'is_correct': False,
   553→            'already_solved': False,
   554→            'exhausted': True,
   555→            'attempts_used': progress.attempts,
   556→            'max_attempts': mcq.max_attempts,
   557→            'correct_choice_ids': correct_ids,
   558→            'explanation': mcq.explanation,
   559→        })
   560→
   561→    progress.attempts += 1
   562→    progress.save(update_fields=['attempts'])
   563→
   564→    # Evaluate correctness
   565→    all_correct = set(MCQChoice.objects.filter(exercise=mcq, is_correct=True).values_list('id', flat=True))
   566→    selected_set = {c.id for c in selected_choices}
   567→
   568→    if mcq.allow_multiple_correct:
   569→        is_correct = (selected_set == all_correct)
   570→    else:
   571→        is_correct = (len(selected_set) == 1 and selected_set == all_correct)
   572→
   573→    # Record attempt via progress helper
   574→    record_submission(
   575→        request.user,
   576→        mcq,
   577→        is_correct,
   578→        points_earned=mcq.points if is_correct else 0,
   579→        answer_data={
   580→            'selected_choice_ids': selected_ids,
   581→            'attempt_number': progress.attempts,
   582→        },
   583→    )
   584→
   585→    response_data = {
   586→        'is_correct': is_correct,
   587→        'already_solved': False,
   588→        'exhausted': False,
   589→        'attempts_used': progress.attempts,
   590→        'max_attempts': mcq.max_attempts,
   591→        'points_earned': 0,
   592→        'per_choice_feedback': {
   593→            str(c.id): c.feedback for c in selected_choices if c.feedback
   594→        },
   595→    }
   596→
   597→    if is_correct:
   598→        progress.is_solved = True
   599→        progress.points_earned = mcq.points
   600→        progress.solved_at = tz.now()
   601→        progress.save()
   602→        response_data['points_earned'] = mcq.points
   603→        response_data['explanation'] = mcq.explanation
   604→        # Notify
   605→        try:
   606→            from notifications.notifications import notify_user
   607→            notify_user(
   608→                request.user,
   609→                title=_("QCM réussi ! 🎉"),
   610→                message=_("Tu as répondu correctement à '%(title)s' (+%(pts)s pts)") % {
   611→                    'title': mcq.title, 'pts': mcq.points
   612→                },
   613→                notification_type='success',
   614→            )
   615→        except Exception:
   616→            pass
   617→    elif mcq.max_attempts > 0 and progress.attempts >= mcq.max_attempts:
   618→        # Max attempts just reached — send answers
   619→        response_data['exhausted'] = True
   620→        response_data['correct_choice_ids'] = list(all_correct)
   621→        response_data['explanation'] = mcq.explanation
   622→    else:
   623→        # Wrong but attempts remain — optionally send explanation
   624→        if getattr(mcq, 'show_explanation_on_wrong', False) and mcq.explanation:
   625→            response_data['explanation'] = mcq.explanation
   626→
   627→    return JsonResponse(response_data)

   676→def _award_and_respond(request, exercise, progress, is_correct, points_max,
   677→                       attempts_count, extra=None, answer_data=None):
   678→    """Common response builder for all server-evaluated exercise types.
   679→    progress is a StudentProgress instance (replaces removed XxxGrade models).
   680→    """
   681→    from .grades import notify_exercise_solved
   682→    from .progress import record_submission
   683→    from django.utils import timezone as tz
   684→
   685→    already_solved = progress.is_solved
   686→    progress.attempts = attempts_count
   687→
   688→    points_earned = 0
   689→    if is_correct and not already_solved:
   690→        progress.is_solved     = True
   691→        progress.points_earned = points_max
   692→        progress.solved_at     = tz.now()
   693→        points_earned          = points_max
   694→        notify_exercise_solved(request.user, exercise.title, points_max)
   695→    progress.save()
   696→
   697→    record_submission(
   698→        request.user, exercise, is_correct,
   699→        points_earned=points_earned, answer_data=answer_data,
   700→    )
   701→
   702→    exhausted = (
   703→        hasattr(exercise, 'max_attempts') and exercise.max_attempts > 0
   704→        and attempts_count >= exercise.max_attempts
   705→    )
   706→    show_explanation = is_correct or exhausted or already_solved
   707→
   708→    data = {
   709→        'is_correct':    is_correct,
   710→        'points_earned': points_earned,
   711→        'already_solved': already_solved,
   712→        'attempts_used': attempts_count,
   713→        'max_attempts':  exercise.max_attempts if hasattr(exercise, 'max_attempts') else 0,
   714→        'explanation': (
   715→            getattr(exercise, 'explanation', '') or ''
   716→        ) if show_explanation else '',
   717→    }
   718→    if extra:
   719→        data.update(extra)
   720→    return JsonResponse(data)

   845→    total          = len(pairs)
   846→    is_correct     = (correct_count == total)

   849→
