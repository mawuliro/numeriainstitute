     1→"""
     2→Management command: seed_python_course
     3→
     4→Creates a complete Python course "De l'Algorithmique à la POO" with:
     5→  - 5 modules
     6→  - 15 lessons
     7→  - ~50 text blocks (Markdown + LaTeX)
     8→  - ~15 sandboxes (Python code)
     9→  - ~20 MCQ exercises (3-4 choices each)
    10→  - ~15 code exercises (with starter code, expected output, tests)
    11→  - ~10 fill-in-the-blank exercises
    12→  - ~10 true/false exercises
    13→
    14→Idempotent: if the course already exists (matched by slug), it is updated
    15→in place rather than duplicated. Run with:
    16→
    17→    python manage.py seed_python_course
    18→
    19→The course is created in 'published' status so it's immediately visible
    20→on the catalogue. Set --draft to keep it as a draft instead.
    21→"""
    22→from django.core.management.base import BaseCommand
    23→from django.db import transaction
    24→
    25→from cours.models import (
    26→    Course, CourseModule, CourseLesson,
    27→    LessonBlock,
    28→    CodeExercise, MCQExercise, MCQChoice,
    29→    FillBlankExercise, TrueFalseExercise,
    30→)
    31→
    32→
    33→class Command(BaseCommand):
    34→    help = 'Seed a complete Python course (Algorithmique → POO) with lessons, blocks, and exercises.'
    35→
    36→    def add_arguments(self, parser):
    37→        parser.add_argument(
    38→            '--draft', action='store_true',
    39→            help='Create the course in draft status (default: published).',
    40→        )
    41→        parser.add_argument(
    42→            '--clean', action='store_true',
    43→            help='Delete the existing course (and all its children) before re-creating.',
    44→        )
    45→
    46→    @transaction.atomic
    47→    def handle(self, *args, **options):
    48→        slug = 'python-algorithmique-poo'
    49→        status = 'draft' if options['draft'] else 'published'
    50→
    51→        if options['clean']:
    52→            deleted, _ = Course.objects.filter(slug=slug).delete()
    53→            if deleted:
    54→                self.stdout.write(self.style.WARNING(f'Deleted existing course ({deleted} related rows).'))
    55→
    56→        course, created = Course.objects.get_or_create(
    57→            slug=slug,
    58→            defaults={
    59→                'title': 'Python : de l\'Algorithmique à la POO',
    60→                'description': (
    61→                    "Un parcours complet pour apprendre Python depuis les premiers "
    62→                    "concepts algorithmiques jusqu'à la programmation orientée objet. "
    63→                    "Pensé pour les apprenants africains et francophones : exemples "
    64→                    "concrets, exercices interactifs, et sandbox Python intégrée."
    65→                ),
    66→                'short_description': (
    67→                    "Apprends Python de zéro à la POO : algorithmique, structures de "
    68→                    "données, récursivité, tri, classes, héritage."
    69→                ),
    70→                'category': 'python',
    71→                'level': 'debutant',
    72→                'language': 'fr',
    73→                'price': 0,
    74→                'is_free': True,
    75→                'status': status,
    76→                'estimated_hours': 40,
    77→            },
    78→        )
    79→        if not created:
    80→            self.stdout.write(self.style.WARNING(f'Course "{course.title}" already exists — updating.'))
    81→            course.status = status
    82→            course.save(update_fields=['status'])
    83→        else:
    84→            self.stdout.write(self.style.SUCCESS(f'Created course "{course.title}".'))
    85→
    86→        # Build each module + lessons + blocks
    87→        for module_data in COURSE_STRUCTURE:
    88→            module = self.upsert_module(course, module_data)
    89→            for lesson_data in module_data['lessons']:
    90→                lesson = self.upsert_lesson(course, module, lesson_data)
    91→                self.upsert_blocks(lesson, lesson_data['blocks'])
    92→
    93→        self.stdout.write(self.style.SUCCESS(
    94→            f'\n✓ Course seeded: {course.title}\n'
    95→            f'  Modules: {CourseModule.objects.filter(course=course).count()}\n'
    96→            f'  Lessons: {CourseLesson.objects.filter(course=course).count()}\n'
    97→            f'  Blocks:  {LessonBlock.objects.filter(course_lesson__course=course).count()}\n'
    98→            f'  Status:  {course.status}\n'
    99→            f'  URL:     /cours/{course.id}/'
   100→        ))
   101→
   102→    def upsert_module(self, course, data):
   103→        module, _ = CourseModule.objects.get_or_create(
   104→            course=course, title=data['title'],
   105→            defaults={'description': data.get('description', ''), 'order': data['order'], 'is_active': True},
   106→        )
   107→        return module
   108→
   109→    def upsert_lesson(self, course, module, data):
   110→        slug = data.get('slug') or data['title'].lower().replace(' ', '-').replace('é', 'e').replace('à', 'a').replace('è', 'e').replace('û', 'u').replace("'", '-')
   111→        lesson, _ = CourseLesson.objects.get_or_create(
   112→            course=course, module=module, title=data['title'],
   113→            defaults={
   114→                'slug': slug,
   115→                'order': data['order'],
   116→                'estimated_minutes': data.get('minutes', 20),
   117→                'is_free_preview': data.get('free_preview', True),
   118→                'is_active': True,
   119→            },
   120→        )
   121→        return lesson
   122→
   123→    def upsert_blocks(self, lesson, blocks_data):
   124→        # Clear existing blocks for this lesson (so re-running updates content cleanly)
   125→        LessonBlock.objects.filter(course_lesson=lesson).delete()
   126→        for idx, block_data in enumerate(blocks_data):
   127→            self.create_block(lesson, block_data, idx)
   128→
   129→    def create_block(self, lesson, data, idx):
   130→        btype = data['type']
   131→        block = LessonBlock(course_lesson=lesson, block_type=btype, order=idx)
   132→        if btype == 'text':
   133→            block.text_content = data['content']
   134→            block.save()
   135→        elif btype == 'sandbox':
   136→            block.sandbox_title = data.get('title', 'Essaie toi-même')
   137→            block.sandbox_initial_code = data.get('code', '# Écris ton code ici\n')
   138→            block.save()
   139→        elif btype == 'mcq':
   140→            ex = self.create_mcq(lesson, data)
   141→            block.mcq_exercise = ex
   142→            block.save()
   143→        elif btype == 'code_exercise':
   144→            ex = self.create_code_exercise(lesson, data)
   145→            block.code_exercise = ex
   146→            block.save()
   147→        elif btype == 'fill_blank':
   148→            ex = self.create_fill_blank(lesson, data)
   149→            block.fill_blank = ex
   150→            block.save()
   151→        elif btype == 'true_false':
   152→            ex = self.create_true_false(lesson, data)
   153→            block.true_false = ex
   154→            block.save()
   155→        else:
   156→            # Unsupported type for this seeder — skip silently
   157→            return
   158→
   159→    # ---------- Exercise factories ----------
   160→
   161→    def create_mcq(self, lesson, data):
   162→        ex = MCQExercise.objects.create(
   163→            course_lesson=lesson,
   164→            title=data['title'],
   165→            question=data['question'],
   166→            instructions=data.get('instructions', ''),
   167→            difficulty=data.get('difficulty', 'easy'),
   168→            points=data.get('points', 5),
   169→            hint=data.get('hint', ''),
   170→            explanation=data.get('explanation', ''),
   171→            order=data.get('order', 0),
   172→            allow_multiple_correct=data.get('multiple', False),
   173→            shuffle_choices=True,
   174→        )
   175→        for i, choice in enumerate(data['choices']):
   176→            MCQChoice.objects.create(
   177→                exercise=ex,
   178→                text=choice['text'],
   179→                is_correct=choice['correct'],
   180→                feedback=choice.get('feedback', ''),
   181→                order=i,
   182→            )
   183→        return ex
   184→
   185→    def create_code_exercise(self, lesson, data):
   186→        return CodeExercise.objects.create(
   187→            course_lesson=lesson,
   188→            title=data['title'],
   189→            instructions=data.get('instructions', ''),
   190→            difficulty=data.get('difficulty', 'easy'),
   191→            points=data.get('points', 10),
   192→            hint=data.get('hint', ''),
   193→            explanation=data.get('explanation', ''),
   194→            order=data.get('order', 0),
   195→            starter_code=data.get('starter', '# Écris ton code ici\n'),
   196→            solution_code=data.get('solution', ''),
   197→            expected_output=data.get('expected_output', ''),
   198→            test_code=data.get('test_code', ''),
   199→            evaluation_mode=data.get('eval_mode', 'exact'),
   200→        )
   201→
   202→    def create_fill_blank(self, lesson, data):
   203→        return FillBlankExercise.objects.create(

   208→            points=data.get('points', 5),
   209→            hint=data.get('hint', ''),
   210→            explanation=data.get('explanation', ''),
   211→            order=data.get('order', 0),
   212→            text_with_blanks=data['text_with_blanks'],
   213→            answers=data['answers'],
   214→            case_sensitive=data.get('case_sensitive', False),
   215→        )
   216→
   217→    def create_true_false(self, lesson, data):
   218→        return TrueFalseExercise.objects.create(
   219→            course_lesson=lesson,
   220→            title=data['title'],
   221→            instructions=data.get('instructions', ''),
   222→            difficulty=data.get('difficulty', 'easy'),
   223→            points=data.get('points', 6),
   224→            hint=data.get('hint', ''),
   225→            explanation=data.get('explanation', ''),
   226→            order=data.get('order', 0),
   227→            statements=data['statements'],
   228→            points_per_statement=data.get('points_per_statement', 2),
   229→        )

   511→| `<` | Strictement inférieur | `3 < 5` → `True` |
   512→| `>` | Strictement supérieur | `5 > 3` → `True` |
   513→| `<=` | Inférieur ou égal | `5 <= 5` → `True` |
   514→| `>=` | Supérieur ou égal | `4 >= 5` → `False` |
   515→
   516→⚠ **Attention** : `=` (affectation) et `==` (comparaison) sont différents !
   517→
   518→## Opérateurs logiques
   519→
   520→Python possède trois opérateurs logiques :
   521→
   522→| Opérateur | Description | Exemple |
   523→|-----------|-------------|---------|
   524→| `and` | ET (les deux vrais) | `True and False` → `False` |
   525→| `or` | OU (au moins un vrai) | `True or False` → `True` |
   526→| `not` | NON (inverse) | `not True` → `False` |
   527→
   528→```python
   529→age = 20
   530→permis = True
   531→print(age >= 18 and permis)   # True — peut conduire
   532→print(age < 18 or not permis) # False
   533→```
   534→

  1720→Toute fonction récursive doit avoir :

  1740→

  1757→

  1763→    if n == 1: return 1

  1772→

  1898→## Tri par sélection
  1899→
  1900→**Idée** : à chaque tour, on trouve le minimum et on l'échange avec l'élément courant.
  1901→
  1902→```python
  1903→def tri_selection(liste):
  1904→    n = len(liste)
  1905→    for i in range(n):

  1907→        min_idx = i
  1908→        for j in range(i + 1, n):
  1909→            if liste[j] < liste[min_idx]:
  1910→                min_idx = j
  1911→        # Échanger
  1912→        liste[i], liste[min_idx] = liste[min_idx], liste[i]
  1913→    return liste
  1914→
  1915→print(tri_selection([5, 2, 8, 1, 9, 3]))
  1916→# [1, 2, 3, 5, 8, 9]
  1917→```

  1920→
  1921→## Tri à bulles
  1922→
  1923→**Idée** : on compare les éléments adjacents et on les échange s'ils sont dans le mauvais ordre. Les plus grands "remontent" comme des bulles.
  1924→

  1926→def tri_bulles(liste):
  1927→    n = len(liste)
  1928→    for i in range(n):
  1929→        for j in range(0, n - i - 1):
  1930→            if liste[j] > liste[j + 1]:
  1931→                liste[j], liste[j + 1] = liste[j + 1], liste[j]
  1932→    return liste
  1933→

  1935→```
  1936→
  1937→**Complexité** : $O(n^2)$.

  1944→def tri_insertion(liste):
  1945→    for i in range(1, len(liste)):
  1946→        cle = liste[i]
  1947→        j = i - 1
  1948→        while j >= 0 and liste[j] > cle:
  1949→            liste[j + 1] = liste[j]
  1950→            j -= 1
  1951→        liste[j + 1] = cle
  1952→    return liste
  1953→
  1954→print(tri_insertion([5, 2, 8, 1, 9, 3]))
  1955→```
  1956→
  1957→**Complexité** : $O(n^2)$ dans le pire cas, $O(n)$ sur une liste déjà triée.
  1958→
  1959→## Tri rapide (Quicksort) — diviser pour régner
  1960→
  1961→**Idée** : choisir un **pivot**, partitionner la liste en deux (éléments ≤ pivot et > pivot), trier récursivement chaque moitié.
  1962→
  1963→```python
  1964→def tri_rapide(liste):
  1965→    if len(liste) <= 1:
  1966→        return liste
  1967→    pivot = liste[0]
  1968→    gauche = [x for x in liste[1:] if x <= pivot]
  1969→    droite = [x for x in liste[1:] if x > pivot]
  1970→    return tri_rapide(gauche) + [pivot] + tri_rapide(droite)
  1971→
  1972→print(tri_rapide([5, 2, 8, 1, 9, 3]))
  1973→```
  1974→
  1975→**Complexité** : $O(n \\log n)$ en moyenne, $O(n^2)$ dans le pire cas.
  1976→
  1977→## Python a déjà tout prévu

  1982→liste = [5, 2, 8, 1, 9, 3]
  1983→print(sorted(liste))      # [1, 2, 3, 5, 8, 9] — nouvelle liste
