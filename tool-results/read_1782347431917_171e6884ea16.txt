    25→    25→def extraire_id_youtube(url: str | None):
    26→    26→    """Extract YouTube video ID from common URL formats."""
    27→    27→    if not url or 'vimeo' in (url or '').lower():
    28→    28→        return None
    29→    29→    url = url.strip()
    30→    30→    if 'youtu.be/' in url:
    31→    31→        return url.split('youtu.be/')[-1].split('?')[0].strip('/')
    32→    32→    if '/shorts/' in url:
    33→    33→        return url.split('/shorts/')[-1].split('?')[0].strip('/')
    34→    34→    if '/embed/' in url:
    35→    35→        return url.split('/embed/')[-1].split('?')[0].strip('/')
    36→    36→    if 'v=' in url:
    37→    37→        return url.split('v=')[-1].split('&')[0].strip()
    38→    38→    return None
    39→    39→

    41→    41→def convertir_url_youtube(url: str | None):
    42→    42→    """Convert YouTube/Vimeo URL to embed URL (display-time only)."""
    43→    43→    if not url:
    44→    44→        return url
    45→    45→
    46→    46→    url = url.strip()
    47→    47→    u = url.lower()
    48→    48→
    49→    49→    # Vimeo
    50→    50→    if 'vimeo.com' in u:
    51→    51→        if '/video/' in u:
    52→    52→            video_id = url.split('/video/')[-1].split('?')[0].split('/')[0]
    53→    53→        else:
    54→    54→            video_id = url.split('vimeo.com/')[-1].split('?')[0].split('/')[0]
    55→    55→        if video_id.isdigit():
    56→    56→            return f'https://player.vimeo.com/video/{video_id}?color=E8A020&title=0&byline=0'
    57→    57→        return url
    58→    58→
    59→    59→    # YouTube
    60→    60→    if 'youtube-nocookie.com/embed/' in u:
    61→    61→        return url
    62→    62→    if 'youtube.com/embed/' in u:
    63→    63→        video_id = url.split('/embed/')[-1].split('?')[0].strip('/')
    64→    64→        return f'https://www.youtube-nocookie.com/embed/{video_id}?rel=0&modestbranding=1'
    65→    65→    if 'youtu.be/' in u:
    66→    66→        video_id = url.split('youtu.be/')[-1].split('?')[0].strip('/')
    67→    67→        return f'https://www.youtube-nocookie.com/embed/{video_id}?rel=0&modestbranding=1'
    68→    68→    if '/shorts/' in u:
    69→    69→        video_id = url.split('/shorts/')[-1].split('?')[0].strip('/')
    70→    70→        return f'https://www.youtube-nocookie.com/embed/{video_id}?rel=0&modestbranding=1'
    71→    71→    if 'youtube.com' in u and 'v=' in url:
    72→    72→        video_id = url.split('v=')[-1].split('&')[0].strip()
    73→    73→        return f'https://www.youtube-nocookie.com/embed/{video_id}?rel=0&modestbranding=1'
    74→    74→    return url

   381→   381→    def get_payload(self, user=None, preview=False):

   383→   383→
   384→   384→        Never includes solution_code or test_code in plain text —
   385→   385→        test_code is base64-encoded for client-side evaluation only.
   386→   386→        """
   387→   387→        import base64
   388→   388→
   389→   389→        data = {
   390→   390→            'id': self.id,
   391→   391→            'type': self.block_type,
   392→   392→            'order': self.order,
   393→   393→        }
   394→   394→
   395→   395→        if self.block_type == 'text':
   396→   396→            data['text_content'] = self.text_content or ''
   397→   397→
   398→   398→        elif self.block_type == 'video':
   399→   399→            data['embed_url'] = self._embed_url(self.video_url)
   400→   400→            data['video_caption'] = self.video_caption or ''
   401→   401→
   402→   402→        elif self.block_type == 'sandbox':
   403→   403→            data['title'] = self.sandbox_title or 'Essaie toi-même'
   404→   404→            data['initial_code'] = self.sandbox_initial_code or ''
   405→   405→
   406→   406→        elif self.block_type == 'code_exercise':
   407→   407→            ex = self.code_exercise
   408→   408→            if ex:
   409→   409→                data['exercise_id'] = ex.id
   410→   410→                data['id'] = ex.id
   411→   411→                data['title'] = ex.title
   412→   412→                data['instructions'] = ex.instructions or ''
   413→   413→                data['difficulty'] = ex.difficulty
   414→   414→                data['points'] = ex.points
   415→   415→                data['hint'] = ex.hint or ''
   416→   416→                data['max_attempts'] = ex.max_attempts
   417→   417→                data['starter_code'] = ex.starter_code or ''
   418→   418→                data['expected_output'] = ex.expected_output or ''
   419→   419→                data['evaluation_mode'] = ex.evaluation_mode
   420→   420→                data['test_code_b64'] = base64.b64encode(
   421→   421→                    (ex.test_code or '').encode()
   422→   422→                ).decode() if ex.test_code else ''
   423→   423→                data['order'] = ex.order
   424→   424→                data['is_solved'] = False
   425→   425→                data['points_earned'] = 0
   426→   426→                data['attempts_used'] = 0
   427→   427→                if user and not preview:
   428→   428→                    from cours.models import StudentProgress, ExerciseAttempt
   429→   429→                    progress = StudentProgress.objects.filter(
   430→   430→                        student=user, exercise_type='code',
   431→   431→                        exercise_id=ex.id
   432→   432→                    ).first()
   433→   433→                    if progress:
   434→   434→                        data['is_solved'] = progress.is_solved
   435→   435→                        data['points_earned'] = progress.points_earned
   436→   436→                    data['attempts_used'] = ExerciseAttempt.objects.filter(
   437→   437→                        exercise_type='code', exercise_id=ex.id, student=user
   438→   438→                    ).count()
   439→   439→
   440→   440→        elif self.block_type == 'mcq':
   441→   441→            ex = self.mcq_exercise
   442→   442→            if ex:
   443→   443→                data['mcq_id'] = ex.id
   444→   444→                data['mcq_title'] = ex.title
   445→   445→                data['title'] = ex.title
   446→   446→                data['question'] = ex.question or ''
   447→   447→                data['hint'] = ex.hint or ''
   448→   448→                data['allow_multiple'] = ex.allow_multiple_correct
   449→   449→                data['shuffle'] = ex.shuffle_choices
   450→   450→                data['max_attempts'] = ex.max_attempts
   451→   451→                data['points'] = ex.points
   452→   452→                data['difficulty'] = ex.difficulty
   453→   453→                data['explanation'] = ''  # hidden until solved/exhausted
   454→   454→                data['is_solved'] = False
   455→   455→                data['points_earned'] = 0
   456→   456→                data['attempts_used'] = 0
   457→   457→                data['correct_ids'] = []
   458→   458→                data['choices'] = [
   459→   459→                    {'id': c.id, 'text': c.text, 'order': c.order}
   460→   460→                    for c in ex.choices.all().order_by('order')
   461→   461→                ]
   462→   462→                if user and not preview:
   463→   463→                    from cours.models import StudentProgress, ExerciseAttempt
   464→   464→                    progress = StudentProgress.objects.filter(
   465→   465→                        student=user, exercise_type='mcq',
   466→   466→                        exercise_id=ex.id
   467→   467→                    ).first()
   468→   468→                    if progress:

   572→   572→    @staticmethod
   573→   573→    def _embed_url(url):
   574→   574→        """Convert YouTube/Vimeo URLs to embeddable iframe URLs."""
   575→   575→        if not url:
   576→   576→            return ''
   577→   577→        u = url.strip()
   578→   578→        if 'youtube.com/watch' in u and 'v=' in u:
   579→   579→            vid = u.split('v=', 1)[1].split('&', 1)[0]
   580→   580→            return f'https://www.youtube.com/embed/{vid}'
   581→   581→        if 'youtu.be/' in u:
   582→   582→            vid = u.split('youtu.be/', 1)[1].split('?', 1)[0]
   583→   583→            return f'https://www.youtube.com/embed/{vid}'
   584→   584→        if 'vimeo.com/' in u and '/embed/' not in u:
   585→   585→            vid = u.rstrip('/').split('/')[-1]
   586→   586→            return f'https://player.vimeo.com/video/{vid}'
   587→   587→        return u
