    39→  init() {
    40→    if (!this.roomCode || !this.wsUrl) {
    41→      this.showError('Le code de réunion ou l’URL WebSocket est manquant.');
    42→      return;
    43→    }
    44→
    45→    this.showStatus('Initialisation de la réunion...', false);

    51→
    52→  checkBrowserSupport() {
    53→    if (!navigator.mediaDevices || !navigator.mediaDevices.getUserMedia) {
    54→      this.showError('WebRTC non supporté dans ce navigateur. Utilisez Chrome, Firefox, Safari ou Edge récent.');
    55→      throw new Error('WebRTC unsupported');
    56→    }
    57→    if (!window.RTCPeerConnection) {
    58→      this.showError('Votre navigateur ne supporte pas RTCPeerConnection.');
    59→      throw new Error('RTCPeerConnection unsupported');
    60→    }
    61→  }

    63→  async getUserMedia() {
    64→    try {
    65→      this.localStream = await navigator.mediaDevices.getUserMedia({ video: true, audio: true });
    66→      this.localVideoElement.srcObject = this.localStream;
    67→      this.localVideoElement.play().catch(() => {});
    68→      this.showStatus('Caméra et micro prêts. Connexion au signal...', false);
    69→    } catch (error) {
    70→      console.warn('Erreur getUserMedia:', error);
    71→      try {
    72→        this.localStream = await navigator.mediaDevices.getUserMedia({ video: false, audio: true });
    73→        this.localVideoElement.srcObject = this.localStream;
    74→        this.localVideoElement.play().catch(() => {});
    75→        this.showStatus('Micro actif, caméra désactivée.', true);
    76→        this.isCameraOff = true;
    77→        this.updateLocalCameraState();
    78→      } catch (secondaryError) {
    79→        console.error('Permis refusés ou absence de média:', secondaryError);
    80→        this.showError('Impossible d’accéder à la caméra et au micro. Vérifiez les permissions du navigateur.');
    81→        this.setButtonDisabled('#btn-camera', true);
    82→        this.setButtonDisabled('#btn-mute', true);
    83→        this.showStatus('Média non disponible.', true);
    84→      }
    85→    }

    88→  connectWebSocket() {
    89→    if (!this.shouldReconnect) {
    90→      return;
    91→    }
    92→    this.showStatus('Connexion WebSocket en cours...', false);
    93→    try {
    94→      this.ws = new WebSocket(this.wsUrl);
    95→      this.ws.addEventListener('open', () => this.onWsOpen());
    96→      this.ws.addEventListener('message', event => this.onWsMessage(event));
    97→      this.ws.addEventListener('close', () => this.onWsClose());
    98→      this.ws.addEventListener('error', () => this.onWsError());
    99→    } catch (error) {
   100→      console.error('Erreur WebSocket:', error);
   101→      this.showError('Impossible d’établir la connexion en temps réel.');
   102→    }
   103→  }
   104→
   105→  onWsOpen() {
   106→    this.showStatus('WebSocket connecté ✓', false);
   107→    this.sendMessage({
   108→      type: 'join',
   109→      peer_id: this.localPeerId,
   110→      username: this.localUserName,
   111→      user_id: CURRENT_USER_ID,
   112→    });
   113→  }
   114→
   115→  onWsMessage(event) {
   116→    let payload;
   117→    try {
   118→      payload = JSON.parse(event.data);
   119→    } catch (error) {
   120→      console.warn('Message WebSocket invalide:', error, event.data);
   121→      return;
   122→    }
   123→    this.handleSignaling(payload);
   124→  }
   125→
   126→  onWsClose() {
   127→    if (!this.shouldReconnect) {
   128→      this.showStatus('Connexion fermée.', true);
   129→      return;
   130→    }
   131→    this.showStatus('Connexion WebSocket perdue. Reconnexion...', true);
   132→    setTimeout(() => this.connectWebSocket(), 3000);
   133→  }
   134→
   135→  onWsError() {
   136→    this.showStatus('Erreur WebSocket détectée.', true);
   137→  }
   138→
   139→  sendMessage(data) {
   140→    if (!this.ws || this.ws.readyState !== WebSocket.OPEN) {
   141→      console.warn('WebSocket non prêt pour envoyer:', data);
   142→      return;
   143→    }
   144→    this.ws.send(JSON.stringify(data));
   145→  }

   147→  async handleSignaling(data) {
   148→    switch (data.type) {
   149→      case 'participants_list':
   150→        await this.processParticipants(data.participants || []);
   151→        this.renderChatHistory(data.chat_history || []);
   152→        break;
   153→      case 'user_joined':
   154→        if (data.peer_id === this.localPeerId) return;
   155→        this.addParticipantToSidebar(data.peer_id, data.username, data.is_host);
   156→        await this.createPeerConnection(data.peer_id, false);
   157→        break;
   158→      case 'user_left':
   159→        this.removeRemoteVideo(data.peer_id);
   160→        this.removeParticipantFromSidebar(data.peer_id);
   161→        break;
   162→      case 'mute_all':
   163→        this.setLocalMute(true, true);
   164→        this.showToast('L’hôte a coupé le micro de tous.');
   165→        break;
   166→      case 'camera_off_all':
   167→        this.setLocalCamera(false, true);
   168→        this.showToast('L’hôte a coupé la caméra de tous.');
   169→        break;
   170→      case 'waiting_list':
   171→        if (IS_HOST) {
   172→          this.renderWaitingList(data.waiting || []);
   173→        }
   174→        break;
   175→      case 'admitted':
   176→        if (data.target === undefined || data.target === this.localPeerId) {
   177→          this.showToast('Vous avez été admis·e en réunion.', false);
   178→          setTimeout(() => {
   179→            window.location.reload();
   180→          }, 800);
   181→        }
   182→        break;
   183→      case 'rejected':
   184→        if (data.target === undefined || data.target === this.localPeerId) {
   185→          this.showError(data.message || 'Votre participation a été refusée.');
   186→          setTimeout(() => {
   187→            window.location.href = '/visio/';
   188→          }, 2000);
   189→        }
   190→        break;
   191→      case 'removed':
   192→        if (data.target === undefined || data.target === this.localPeerId) {
   193→          this.showToast(data.message || 'Vous avez été retiré·e par l’hôte.', true);
   194→          setTimeout(() => this.leaveMeeting(), 1200);
   195→        }
   196→        break;
   197→      case 'offer':
   198→        if (data.target !== this.localPeerId) return;
   199→        await this.receiveOffer(data);
   200→        break;
   201→      case 'answer':
   202→        if (data.target !== this.localPeerId) return;
   203→        await this.receiveAnswer(data);
   204→        break;
   205→      case 'ice_candidate':
   206→        if (data.target !== this.localPeerId) return;
   207→        await this.addIceCandidate(data);
   208→        break;
   209→      case 'chat':
   210→        this.displayChatMessage(data.sender, data.message, false);
   211→        break;
   212→      case 'raise_hand':
   213→        this.updateParticipantState(data.peer_id, 'hand');
   214→        break;
   215→      case 'mute_status':
   216→        this.updateParticipantState(data.peer_id, 'mute', data.value);
   217→        break;
   218→      case 'camera_status':
   219→        this.updateParticipantState(data.peer_id, 'camera', data.value);
   220→        break;
   221→      case 'meeting_ended':
   222→        this.showToast('La réunion a été terminée par l’hôte.', true);
   223→        setTimeout(() => this.leaveMeeting(), 2500);
   224→        break;
   225→      case 'error':
   226→        this.showError(data.message || 'Erreur de réunion.');
   227→        break;
   228→      default:
   229→        console.debug('Signal inconnu reçu:', data);
   230→        break;
   231→    }
   232→  }

   234→  async processParticipants(participants) {
   235→    if (!this.participantsList) return;
   236→    this.participantsList.innerHTML = '';
   237→    const remoteParticipants = [];
   238→
   239→    participants.forEach(item => {
   240→      if (item.peer_id === this.localPeerId) return;
   241→      this.addParticipantToSidebar(item.peer_id, item.username, item.is_host);
   242→      remoteParticipants.push(item);
   243→    });
   244→    this.updateVideoCount(remoteParticipants.length + 1);
   245→
   246→    await Promise.all(remoteParticipants.map(item => this.createPeerConnection(item.peer_id, true)));
   247→  }

   317→  async createPeerConnection(peerId, initiator) {
   318→    if (this.peerConnections[peerId]) {
   319→      return this.peerConnections[peerId];
   320→    }
   321→
   322→    const configuration = {
   323→      iceServers: [
   324→        { urls: 'stun:stun.l.google.com:19302' },
   325→        { urls: 'stun:stun1.l.google.com:19302' },
   326→        { urls: 'stun:stun2.l.google.com:19302' },
   327→      ],
   328→    };
   329→    const pc = new RTCPeerConnection(configuration);
   330→    this.peerConnections[peerId] = pc;
   331→
   332→    pc.onicecandidate = event => {
   333→      if (event.candidate) {
   334→        this.sendMessage({
   335→          type: 'ice_candidate',
   336→          target: peerId,
   337→          candidate: event.candidate,
   338→        });
   339→      }
   340→    };
   341→
   342→    pc.ontrack = event => {
   343→      this.addRemoteVideo(peerId, event.streams[0], 'Participant');
   344→    };
   345→
   346→    pc.onconnectionstatechange = () => {
   347→      const state = pc.connectionState;
   348→      const tile = document.querySelector(`#remote-tile-${peerId}`);
   349→      if (tile) {
   350→        tile.dataset.connection = state;
   351→      }
   352→      if (state === 'failed' || state === 'disconnected') {
   353→        this.showToast(`Connexion perdue avec ${peerId}.`, true);
   354→      }
   355→    };
   356→
   357→    if (this.localStream) {
   358→      if (this.isScreenSharing && this.screenStream) {
   359→        const sharedVideoTrack = this.screenStream.getVideoTracks()[0];
   360→        if (sharedVideoTrack) {
   361→          pc.addTrack(sharedVideoTrack, this.screenStream);
   362→        }
   363→        this.localStream.getAudioTracks().forEach(track => pc.addTrack(track, this.localStream));
   364→      } else {
   365→        this.localStream.getTracks().forEach(track => pc.addTrack(track, this.localStream));
   366→      }
   367→    }
   368→
   369→    if (initiator) {
   370→      await this.createOffer(peerId);
   371→    }
   372→
   373→    return pc;
   374→  }

   376→  async createOffer(peerId) {
   377→    const pc = this.peerConnections[peerId];
   378→    if (!pc) return;
   379→
   380→    try {
   381→      const offer = await pc.createOffer();
   382→      await pc.setLocalDescription(offer);
   383→      this.sendMessage({
   384→        type: 'offer',
   385→        target: peerId,
   386→        sdp: offer.sdp,
   387→      });
   388→    } catch (error) {
   389→      console.error('Erreur création offre:', error);
   390→      this.showError('Impossible de créer l’offre WebRTC.');
   391→    }
   392→  }
   393→
   394→  async receiveOffer(data) {
   395→    const peerId = data.from;
   396→    if (!peerId) return;
   397→    const pc = await this.createPeerConnection(peerId, false);
   398→    try {
   399→      await pc.setRemoteDescription({ type: 'offer', sdp: data.sdp });
   400→      const answer = await pc.createAnswer();
   401→      await pc.setLocalDescription(answer);
   402→      this.sendMessage({
   403→        type: 'answer',
   404→        target: peerId,
   405→        sdp: answer.sdp,
   406→      });
   407→    } catch (error) {
   408→      console.error('Erreur réception offre:', error);
   409→      this.showError('Impossible de traiter l’offre entrante.');
   410→    }
   411→  }
   412→
   413→  async receiveAnswer(data) {
   414→    const peerId = data.from;
   415→    const pc = this.peerConnections[peerId];
   416→    if (!pc) return;
   417→    try {
   418→      await pc.setRemoteDescription({ type: 'answer', sdp: data.sdp });
   419→    } catch (error) {
   420→      console.error('Erreur réception réponse:', error);
   421→      this.showError('Impossible de traiter la réponse WebRTC.');
   422→    }
   423→  }

   430→      await pc.addIceCandidate(data.candidate);

   432→      console.warn('Impossible d’ajouter ICE candidate:', error);

   485→  setLocalMute(muted, notify = false) {
   486→    if (!this.localStream) return;

   492→      button.textContent = this.isMuted ? '🔇 Micro coupé' : '🎤 Micro actif';
   493→    }

   499→  setLocalCamera(enabled, notify = false) {
   500→    if (!this.localStream) return;
   501→    this.isCameraOff = !enabled;
   502→    this.localStream.getVideoTracks().forEach(track => track.enabled = enabled);
   503→    this.updateLocalCameraState();
   504→    const button = document.querySelector('#btn-camera');
   505→    if (button) {
   506→      button.classList.toggle('bg-red-600', this.isCameraOff);
   507→      button.textContent = this.isCameraOff ? '🚫 Vidéo coupée' : '📷 Vidéo active';
   508→    }
   509→    if (!enabled && this.isScreenSharing) {
   510→      this.stopScreenShare();
   511→    }
   512→    if (notify) {
   513→      this.sendMessage({ type: 'camera_status', peer_id: this.localPeerId, value: enabled });
   514→    }
   515→  }

   517→  async startScreenShare() {
   518→    if (this.isScreenSharing) {
   519→      return this.stopScreenShare();
   520→    }
   521→    if (!navigator.mediaDevices || !navigator.mediaDevices.getDisplayMedia) {
   522→      this.showToast('Partage d’écran non supporté par votre navigateur.', true);
   523→      return;
   524→    }
   525→
   526→    try {
   527→      const screenStream = await navigator.mediaDevices.getDisplayMedia({ video: true });
   528→      const screenTrack = screenStream.getVideoTracks()[0];
   529→      const cameraTrack = this.localStream?.getVideoTracks()[0];
   530→
   531→      if (!screenTrack) {
   532→        throw new Error('Aucune piste d’écran disponible.');
   533→      }
   534→
   535→      this.isScreenSharing = true;
   536→      this.screenStream = screenStream;
   537→      this.screenShareTrack = screenTrack;
   538→      document.querySelector('#btn-screen')?.classList.add('bg-teal-500');
   539→      document.querySelector('#btn-screen').textContent = '🛑 Arrêter le partage';

   744→        this.leaveMeeting();

   751→    }

   797→  toggleRecording() {
   798→    if (!this.localStream) {
   799→      this.showToast('Aucun flux disponible pour l’enregistrement.', true);
   800→      return;
   801→    }
   802→    if (this.mediaRecorder && this.mediaRecorder.state === 'recording') {
   803→      this.mediaRecorder.stop();
   804→      return;
   805→    }
   806→    try {
   807→      this.mediaRecorder = new MediaRecorder(this.localStream);
   808→      this.recordingChunks = [];
   809→      this.mediaRecorder.addEventListener('dataavailable', event => {
   810→        if (event.data.size > 0) {
   811→          this.recordingChunks.push(event.data);
   812→        }
   813→      });
   814→      this.mediaRecorder.addEventListener('stop', () => {
   815→        const blob = new Blob(this.recordingChunks, { type: 'video/webm' });
   816→        const url = URL.createObjectURL(blob);
   817→        const a = document.createElement('a');
   818→        a.href = url;
   819→        a.download = 'numeria-reunion.webm';
   820→        a.click();
   821→        URL.revokeObjectURL(url);
   822→        this.showToast('Enregistrement disponible en téléchargement.');
   823→      });
   824→      this.mediaRecorder.start();
   825→      this.showToast('Enregistrement démarré.');
   826→    } catch (error) {
   827→      console.error('MediaRecorder error:', error);
   828→      this.showToast('Impossible de démarrer l’enregistrement.', true);
   829→    }
   830→  }

   832→  getCsrfToken() {
   833→    const name = 'csrftoken';
   834→    const cookies = document.cookie.split(';');
   835→    for (const cookie of cookies) {
   836→      const [key, value] = cookie.trim().split('=');
   837→      if (key === name) {
   838→        return decodeURIComponent(value);
   839→      }
   840→    }
   841→    return '';
   842→  }
